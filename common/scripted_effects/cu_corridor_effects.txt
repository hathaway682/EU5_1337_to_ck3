# Province scope. The native Sahara modifier uses the same permanent form:
# add_province_modifier without a duration has no scheduled expiry.
cu_corridor_mark_province_effect = {
    if = {
        limit = { NOT = { has_province_modifier = cu_corridor_land_modifier } }
        add_province_modifier = cu_corridor_land_modifier
    }
}

# New-game province scope. Use the CURRENT county capital, not the old list.
# After merging into an ordinary county, its non-capital corridor land stays empty.
cu_corridor_initialize_province_holding_effect = {
    cu_corridor_mark_province_effect = yes
    if = {
        limit = { is_county_capital = yes }
        if = {
            limit = { NOT = { has_holding_type = wilderness_holding } }
            set_holding_type = wilderness_holding
        }
    }
    else_if = {
        limit = { has_holding = yes }
        remove_holding = yes
    }
}
