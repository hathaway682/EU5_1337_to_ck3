# CHARACTER scope: grant charts of both Americas to the holder and the top liege.
eu5_award_american_charts_effect = {
 every_held_title = { limit = { tier >= tier_kingdom } set_variable = eu5_charted_north_america }
 top_liege ?= { every_held_title = { limit = { tier >= tier_kingdom } set_variable = eu5_charted_north_america } }
 every_held_title = { limit = { tier >= tier_kingdom } set_variable = eu5_charted_south_america }
 top_liege ?= { every_held_title = { limit = { tier >= tier_kingdom } set_variable = eu5_charted_south_america } }
}
