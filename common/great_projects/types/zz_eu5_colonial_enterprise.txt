 is_shown = {
  is_landed = yes
  highest_held_title_tier >= tier_kingdom
  NOT = { government_has_flag = government_is_wilderness }
 }

    custom_tooltip = {
     text = eu5_colonial_target_valid_tt
     eu5_colonial_project_target_trigger = { RULER = root }

  }

   any_contribution = {
    contribution_id = settlement
    contribution_is_funded = yes
    NOT = {
     contributor ?= {
      is_alive = yes
      OR = {
       AND = { is_landless_merchant = yes suzerain ?= scope:owner }
       AND = { is_landed = yes OR = { this = scope:owner is_vassal_or_below_of = scope:owner } }
      }
     }
    }
   }
  }
  scope:province.duchy ?= {
   any_de_jure_county = { is_wilderness_county_trigger = yes }
  }
 }

  custom_tooltip = {
   text = eu5_colonial_settlement_outcome_tt
   every_contribution = {
    limit = { contribution_id = settlement contribution_is_funded = yes }
    contributor ?= { eu5_colonial_enterprise_complete_effect = yes }
   }
  }
 }

  settlement = {
   contributor_is_valid = {
    custom_tooltip = {
     text = eu5_colonial_settlement_investor_tt

      AND = { is_landless_merchant = yes suzerain ?= scope:owner scope:owner ?= { culture ?= { has_innovation = innovation_eu5_colonial_charters } } }

       is_landed = yes
       OR = { this = scope:owner is_vassal_or_below_of = scope:owner }
       scope:province.duchy ?= {
        any_de_jure_county = {
         is_wilderness_county_trigger = yes
         any_title_to_title_neighboring_county = {
          holder ?= {
           OR = {
            this = root
            AND = {
             root = { NOT = { this = scope:owner } }
             is_vassal_or_below_of = root
            }
           }
          }
         }
        }
       }
      }
     }
    }
   }
   allowed_contributor_filter = { owner = yes vassals = yes tributaries = yes }
   cost = { gold = 100 }
   is_required = yes
   contributor_cooldown = 0
   on_contribution_funded = { great_project_notify_contribution = yes }
   ai_will_do = { value = 100 }
   ai_check_interval = 3
  }
  supplies = {
   contributor_is_valid = { OR = { this = scope:owner is_vassal_or_below_of = scope:owner AND = { is_landless_merchant = yes suzerain ?= scope:owner scope:owner ?= { culture ?= { has_innovation = innovation_eu5_colonial_charters } } } } }
   cost = { gold = 100 }
   is_required = no

   on_contribution_funded = { great_project_notify_contribution = yes }

   ai_will_do = { value = 50 }
   ai_check_interval = 12
  }
  settlers = {
   contributor_is_valid = { OR = { this = scope:owner is_vassal_or_below_of = scope:owner AND = { is_landless_merchant = yes suzerain ?= scope:owner scope:owner ?= { culture ?= { has_innovation = innovation_eu5_colonial_charters } } } } }
   cost = { gold = 100 }

   ai_will_do = { value = 50 }
   ai_check_interval = 12
  }
  defenses = {
   contributor_is_valid = { OR = { this = scope:owner is_vassal_or_below_of = scope:owner AND = { is_landless_merchant = yes suzerain ?= scope:owner scope:owner ?= { culture ?= { has_innovation = innovation_eu5_colonial_charters } } } } }

  }
