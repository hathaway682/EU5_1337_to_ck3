phx_malbork_high_castle = {
	icon = {
		reference = "gfx/interface/icons/great_projects/great_wall.dds"
	}
	illustration = {
		reference = "gfx/interface/illustrations/decisions/pam_grand_project_cathedral.dds"
	}
	name = {
		desc = great_project_type_phx_malbork_high_castle
	}

	is_shown = {
		is_landed = yes
		has_title = title:d_prussia
		NOT = { has_global_variable = phx_malbork_high_castle_completed }
	}

	province_filter = all
	owner = founder_primary_title_owner

	can_start_planning = {
		is_landed = yes
		has_title = title:d_prussia
		NOT = { is_planning_great_project = phx_malbork_high_castle }
		NOT = { has_global_variable = phx_malbork_high_castle_completed }
		OR = {
			completely_controls = title:c_upper_prussia
			completely_controls = title:c_danzig
		}
	}

	is_location_valid = {
		scope:province = {
			is_county_capital = yes
			county = {
				custom_tooltip = {
					text = phx_malbork_target_valid_tt
					OR = {
						this = title:c_upper_prussia
						this = title:c_danzig
					}
					holder = root
					NOT = { has_county_modifier = phx_teu_marienburg_fortress_modifier }
				}
			}
		}
	}

	cost = {
		gold = 250
		prestige = 150
	}
	construction_time = 1095
	contributor_cooldown = 0
	allowed_contributor_filter = {
		owner = yes
		vassals = yes
		allies = yes
	}

	is_valid = {
		scope:owner ?= {
			is_landed = yes
			has_title = title:d_prussia
		}
	}

	on_complete = {
		set_global_variable = {
			name = phx_malbork_high_castle_completed
			value = yes
		}
		scope:province ?= {
			county = {
				add_county_modifier = {
					modifier = phx_teu_marienburg_fortress_modifier
					years = 100
				}
				change_prosperity_progress = 150
			}
		}
		scope:owner ?= {
			add_prestige = 500
			add_piety = 250
			if = { limit = { is_valid_for_legitimacy_change = yes } add_legitimacy = 150 }
			hidden_effect = {
				trigger_event = phx_ne.0105
			}
		}
	}

	on_cancel = { }

	project_contributions = {
		outer_bailey = {
			contributor_is_valid = {
				OR = {
					this = scope:owner
					is_vassal_or_below_of = scope:owner
					is_allied_to = scope:owner
				}
			}
			allowed_contributor_filter = {
				owner = yes
				vassals = yes
				allies = yes
			}
			cost = {
				gold = 200
			}
			is_required = yes
			contributor_cooldown = 0
			on_contribution_funded = {
				great_project_notify_contribution = yes
				add_prestige = 150
			}
			ai_will_do = { value = 100 }
			ai_check_interval = 6
		}
		grand_refectory = {
			contributor_is_valid = {
				OR = {
					this = scope:owner
					is_vassal_or_below_of = scope:owner
					is_allied_to = scope:owner
				}
			}
			allowed_contributor_filter = {
				owner = yes
				vassals = yes
				allies = yes
			}
			cost = {
				gold = 150
			}
			is_required = no
			contributor_cooldown = 0
			on_contribution_funded = {
				great_project_notify_contribution = yes
				add_prestige = 100
			}
			on_complete = {
				scope:province ?= {
					county = {
						change_prosperity_progress = 75
					}
				}
			}
			ai_will_do = { value = 60 }
			ai_check_interval = 12
		}
		chapel_of_saint_mary = {
			contributor_is_valid = {
				OR = {
					this = scope:owner
					is_vassal_or_below_of = scope:owner
					is_allied_to = scope:owner
				}
			}
			allowed_contributor_filter = {
				owner = yes
				vassals = yes
				allies = yes
			}
			cost = {
				piety = 150
			}
			is_required = no
			contributor_cooldown = 0
			on_contribution_funded = {
				great_project_notify_contribution = yes
				add_piety = 100
			}
			on_complete = {
				scope:owner ?= {
					add_piety = 300
				}
			}
			ai_will_do = { value = 60 }
			ai_check_interval = 12
		}
	}

	ai_will_do = { value = 50 }
	ai_check_interval = 24
}
