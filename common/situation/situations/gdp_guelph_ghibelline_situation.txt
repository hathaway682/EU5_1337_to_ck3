guelph_ghibelline = {
	situation_group_type = minor
	is_unique = yes
	keep_full_history = yes
	map_mode = participant_groups_inside_realm

	##################################################
	# Regions
	##################################################

	sub_regions = {
		core = {
			icon = "gfx/interface/icons/situation_types/guelph_ghibelline.dds"
		}
	}

	##################################################
	# On Actions
	##################################################

	on_start = {
		root.situation_sub_region:core = { 
			add_geographical_region = gdp_guelph_ghibelline_region
		}
		#Hand out the GG story to relevant parties
		if = {
			limit = { game_start_date = 1178.10.1 } #AI Heinrich & Pope has other things to tend to
			random_situation_participant = {
				limit = {
					has_variable = ghibelline_leader
					is_ai = yes
				}
				create_story = story_guelph_ghibelline
			}
			random_situation_participant = {
				limit = {
					has_variable = guelph_leader
					is_ai = yes
				}
				create_story = story_guelph_ghibelline
			}
		}
		else = {
			set_global_variable = {
				name = active_guelph_ghibelline
				value = 0
			}
		}
		#8~ members (4 Guelph, 4 Ghibelline) in 1178, 10 in 1066
		while = {
			limit = {
				exists = global_var:active_guelph_ghibelline
				trigger_if = {
					limit = { game_start_date = 1178.10.1 }
					global_var:active_guelph_ghibelline < 6 #2 leaders + 4 Ghibellines
				}
				trigger_else = { global_var:active_guelph_ghibelline < 5 }	# 0 leaders + 5 Ghibellines
				any_situation_participant = {
					is_available_quick = { ai = yes adult = yes }
					NOT = { has_variable = striving_guelph_ghibelline }
					top_participant_group:guelph_ghibelline ?= { participant_group_type = ghibelline_member }
				}
			}
			random_situation_participant = {
				limit = {
					is_available_quick = {
						ai = yes
						adult = yes
					}
					top_participant_group:guelph_ghibelline ?= { participant_group_type = ghibelline_member }
					NOT = { has_variable = striving_guelph_ghibelline }
				}
				weight = {
					base = 10
					modifier = {
						save_temporary_scope_as = temp_check
						add = scope:temp_check.intrigue
					}
				}
				create_story = story_guelph_ghibelline
			}
		}
		while = {
			limit = {
				exists = global_var:active_guelph_ghibelline
				global_var:active_guelph_ghibelline < 10
				any_situation_participant = {
					is_available_quick = { ai = yes adult = yes }
					NOT = { has_variable = striving_guelph_ghibelline }
					top_participant_group:guelph_ghibelline ?= { participant_group_type = guelph_member }
				}
			}
			random_situation_participant = {
				limit = {
					is_available_quick = {
						ai = yes
						adult = yes
					}
					top_participant_group:guelph_ghibelline ?= { participant_group_type = guelph_member }
					NOT = { has_variable = striving_guelph_ghibelline }
				}
				weight = {
					base = 10
					modifier = {
						save_temporary_scope_as = temp_check
						add = scope:temp_check.intrigue
					}
				}
				create_story = story_guelph_ghibelline
			}
		}
		#The Lombard League in 1178!
		if = {
			limit = { game_start_date = 1178.10.1 }
			character:213600 = { #Venice
				add_character_modifier = lombard_founder_modifier
				create_confederation = {
					name = lombard_league_name
					type = confederation_type:lombard_league
				}
				scope:new_confederation ?= { set_coa = lombard_league }
				add_to_list = lombard_league_member_list
			}
			character:cisalpine0204 = { add_to_list = lombard_league_member_list } #Verona
			character:223597 = { add_to_list = lombard_league_member_list } #Padua / Polesine
			character:cisalpine0208 = { add_to_list = lombard_league_member_list } #Treviso
			character:asinelli_1 = { add_to_list = lombard_league_member_list } #Bologna
			character:cisalpine0153 = { add_to_list = lombard_league_member_list } #Ferrara
			character:cisalpine0158 = { add_to_list = lombard_league_member_list } #Modena
			character:bonacolsi_1 = { add_to_list = lombard_league_member_list } #Mantua / Guastalla
			character:carpineto_1 = { add_to_list = lombard_league_member_list } #Cremona
			character:osa_1 = { add_to_list = lombard_league_member_list } #Brescia
			character:71264 = { add_to_list = lombard_league_member_list } #Novara
			character:beccaria_1 = { add_to_list = lombard_league_member_list } #Tortona
			character:cisalpine0778 = { add_to_list = lombard_league_member_list } #Parma / Piacenza
			character:91416 = { add_to_list = lombard_league_member_list } #Genoa
			character:213007 = { add_to_list = lombard_league_member_list } #Milano
			every_in_list = {
				list = lombard_league_member_list
				save_temporary_scope_as = temp_ll_member
				scope:new_confederation = {
					if = {
						limit = {
							NOT = { 
								any_confederation_member = { this = scope:temp_ll_member }
							}
						}
						add_confederation_member = scope:temp_ll_member
					}
				}
				#Lombard Leaguers get a truce with the HRE Kaiser up until 1183
				add_truce_both_ways = {
					character = character:212501
					years = 5
					name = TRUCE_TREATY_OF_VENICE
					result = victory
				}
			}
		}
		if = {
			limit = { game_start_date = 1066.9.15 }
			character:7804 ?= { add_to_list = gg_starting_involvement_list } #Vitale Faliero
			character:7757 ?= { add_to_list = gg_starting_involvement_list } #Matilda of Canossa
			character:11030 ?= { add_to_list = gg_starting_involvement_list } #Godfrey the Bearded
			character:1118 ?= { add_to_list = gg_starting_involvement_list } #Azzo II d'Este
			character:1530 ?= { add_to_list = gg_starting_involvement_list } #Welf IV
			character:112600 ?= { add_to_list = gg_starting_involvement_list } #Otto of Nordheim
			character:1610 ?= { add_to_list = gg_starting_involvement_list } #Berthold of Zahringen
		}
		else_if = {
			limit = { game_start_date = 1178.10.1 }
			character:213110 ?= { add_to_list = gg_starting_involvement_list } #Humbert III of Savoy
			character:guidi_1 ?= { add_to_list = gg_starting_involvement_list } #Guido Guerra III
			character:dalesmanni_1 ?= { add_to_list = gg_starting_involvement_list } #Speronella Dalesmanni
			character:213100 ?= { add_to_list = gg_starting_involvement_list } #Konrad von Urslingen
			character:71810 ?= { add_to_list = gg_starting_involvement_list } #Pope Alexander III
			character:212600 ?= { add_to_list = gg_starting_involvement_list } #Henry the Lion
			character:213520 ?= { add_to_list = gg_starting_involvement_list } #William V of Montferrat
			character:212960 ?= { add_to_list = gg_starting_involvement_list } #Otto I of Wittelsbach
			character:223597 ?= { add_to_list = gg_starting_involvement_list } #Obizzo I d'Este
			character:cisalpine0208 ?= { add_to_list = gg_starting_involvement_list } #Ezzelino I da Romano
			character:213600 ?= { add_to_list = gg_starting_involvement_list } #Doge Orio Mastropiero
		}
		every_situation_participant = {
			limit = {
				has_variable = gg_involvement
				OR = {
					has_variable = guelph_leader
					has_variable = ghibelline_leader
				}
			}
			gg_add_involvement_effect = { AMOUNT = gg_involvement_minor_value }
		}
		every_situation_participant = {
			limit = {
				has_variable = gg_involvement
				is_in_list = gg_starting_involvement_list
				NOR = {
					has_variable = guelph_leader
					has_variable = ghibelline_leader
				}
			}
			gg_add_involvement_effect = { AMOUNT = gg_involvement_miniscule_value }
		}
		#Splash screen
		every_situation_participant = {
			if = {
				limit = {
					is_ai = no
					OR = {
						has_variable = ghibelline_leader
						has_variable = guelph_leader
						any_held_title = {
							OR = {
								this = title:k_venice
								title_province = { geographical_region = gdp_guelph_ghibelline_northern_italy_region }
							}
						}
						top_overlord ?= {
							any_held_title = {
								OR = {
									this = title:k_venice
									title_province = { geographical_region = gdp_guelph_ghibelline_northern_italy_region }
								}
							}
						}
					}
				}
				trigger_event = gdp_guelph_ghibelline.0500 # Introduction to the Guelphs & Ghibellines
			}
		}
	}

	on_end = {
		every_situation_participant = {
			remove_variable = guelph_leader
			remove_variable = ghibelline_leader
			remove_variable = ghibelline_member_override
			remove_variable = guelph_member_override
			remove_variable = changed_to_ghibelline
			remove_variable = changed_to_guelph
			remove_variable = schemed_to_ghibelline
			remove_variable = schemed_to_guelph
			remove_variable = undecided_override
			remove_variable = gg_involvement
		}
	}

	on_monthly = {
		###Historical stuff
		#1075 Investiture Crisis / Dictatus Papae
		if = {
			limit = {
				situation_current_phase = in_nomine_domini
				current_year = 1075
				NOT = { situation_current_phase = dictatus_papae }
			}
			situation:guelph_ghibelline = {
				if = {
					limit = {
						NOT = { situation_current_phase = dictatus_papae }
					}
					every_situation_participant = {
						limit = { is_ai = no }
						send_interface_message = {
							type = msg_guelph_ghibelline_dictatus_papae_neutral_text
							title = dictatus_papae
							desc = change_situation_phase_to_dictatus_papae_tt
						}
					}
					situation_top_sub_region = { change_phase = dictatus_papae }
				}
			}
		}
		#1123 Concordat of Worms
		if = {
			limit = {
				situation_current_phase = dictatus_papae
				NOT = { has_variable = had_concordat_of_worms }
				current_year >= 1123
			}
			random_list = {
				50 = {
					trigger = {
						trigger_if = {
							limit = { has_variable = concordat_of_worms_chance_var }
							var:concordat_of_worms_chance_var < 3
						}
						trigger_else = { always = yes }
					}
					increment_variable_effect = {
						VAR = concordat_of_worms_chance_var
						VAL = 1
					}
				}
				50 = {
					#Uh-oh, at least one leader is a player
					if = {
						limit = {
							any_situation_participant = {
								OR = {
									has_variable = guelph_leader
									has_variable = ghibelline_leader
								}
								is_ai = no
							}
						}
						if = {
							limit = {
								any_situation_participant = {
									has_variable = guelph_leader
									is_ai = no
								}
							}
							#Guelphs decide
							random_situation_participant = {
								limit = { has_variable = guelph_leader }
								trigger_event = gdp_guelph_ghibelline.6500
							}
						}
						else = {
							#Ghibellines decide
							random_situation_participant = {
								limit = { has_variable = ghibelline_leader }
								trigger_event = gdp_guelph_ghibelline.6505
							}
						}
					}
					else = {
						#Are any of the leaders at war with someone from the other side?
						if = {
							limit = {
								OR = {
									any_situation_participant = {
										has_variable = guelph_leader
										save_temporary_scope_as = guelph_leader_check
										any_character_war = {
											OR = {
												AND = {
													any_war_attacker = { this = scope:guelph_leader_check }
													any_war_defender = { is_ghibelline_trigger = yes }
												}
												AND = {
													any_war_attacker = { is_ghibelline_trigger = yes }
													any_war_defender = { this = scope:guelph_leader_check }
												}
											}
										}
									}
									any_situation_participant = {
										has_variable = ghibelline_leader
										save_temporary_scope_as = ghibelline_leader_check
										any_character_war = {
											OR = {
												AND = {
													any_war_attacker = { this = scope:ghibelline_leader_check }
													any_war_defender = { is_guelph_trigger = yes }
												}
												AND = {
													any_war_attacker = { is_guelph_trigger = yes }
													any_war_defender = { this = scope:ghibelline_leader_check }
												}
											}
										}
									}
								}
							}
							fulfill_concordat_of_worms_effect = { NEW_PHASE = agitation }
						}
						else = {
							fulfill_concordat_of_worms_effect = { NEW_PHASE = repose }
						}
						set_variable = had_concordat_of_worms
					}	
				}
			}
		}
		#1167 The Lombard league forms
		if = {
			limit = {
				NOT = { has_global_variable = lombard_league_created }
				current_year >= 1167
				#The Kaiser needs to exist and be the leader
				any_situation_participant = { 
					has_variable = ghibelline_leader
					primary_title = title:e_hre
				}
			}
			random_list = {
				50 = {
					trigger = {
						trigger_if = {
							limit = { has_variable = lombard_league_chance_var }
							var:lombard_league_chance_var < 5
						}
						trigger_else = { always = yes }
					}
					increment_variable_effect = {
						VAR = lombard_league_chance_var
						VAL = 1
					}
				}
				50 = {
					#Checks'n'balances - start looking for the first-hand Lombard leader - Venice

						limit = {
							title:k_venice ?= {
								holder ?= { 
									is_either_hre_vassal_or_independent_king_trigger = yes
									is_ai = yes
								}
							}
						}
						title:k_venice.holder ?= {
							save_scope_as = league_creator
							save_scope_value_as = {
								name = league_creator_title
								value = flag:venice
							}
							if = {
								limit = { scope:league_creator_title = flag:venice }
								#Error Suppressant
							}
							#AI gets set up automatically
							create_lombard_league_effect = yes
							gg_involvement_decision_effect = yes
						}
					}
					#Alright, no Venice - Verona?
					else_if = {
						limit = {
							NOR = {
								has_global_variable = lombard_league_created
								exists = scope:new_confederation
							}
						}
						try_to_set_up_lombard_league_with_effect = { TITLE = title:c_verona }
					}
					#Ok, no Verona - what about Padua?
					else_if = {
						limit = {
							NOR = {
								has_global_variable = lombard_league_created
								exists = scope:new_confederation
							}
						}
						try_to_set_up_lombard_league_with_effect = { TITLE = title:c_padua }
					}
					#Treviso?
					else_if = {
						limit = {
							NOR = {
								has_global_variable = lombard_league_created
								exists = scope:new_confederation
							}
						}
						try_to_set_up_lombard_league_with_effect = { TITLE = title:c_treviso }
					}
					#Bologna?
					else_if = {
						limit = {
							NOR = {
								has_global_variable = lombard_league_created
								exists = scope:new_confederation
							}
						}
						try_to_set_up_lombard_league_with_effect = { TITLE = title:c_bologna }
					}
					#Ferrara?
					else_if = {

							NOR = {

								exists = scope:new_confederation
							}
						}
						try_to_set_up_lombard_league_with_effect = { TITLE = title:c_ferrara }
					}
					#Modena?
					else_if = {
						limit = {
							NOR = {
								has_global_variable = lombard_league_created
								exists = scope:new_confederation
							}
						}
						try_to_set_up_lombard_league_with_effect = { TITLE = title:c_modena }
					}
					#Mantua?
					else_if = {

								exists = scope:new_confederation
							}
						}
						try_to_set_up_lombard_league_with_effect = { TITLE = title:c_mantua }
					}
					#Cremona?
					else_if = {
						limit = {
							NOR = {
								has_global_variable = lombard_league_created
								exists = scope:new_confederation
							}
						}
						try_to_set_up_lombard_league_with_effect = { TITLE = title:c_cremona }
					}
					#Brescia?
					else_if = {

							NOR = {

							}

		#A leader can drop out of the struggle without dying or losing their land

			}

				ordered_situation_group_participant = {
					limit = {
						NOT = {
							any_held_title = { this = title:k_papal_state }

				limit = { has_variable = ghibelline_leader }

			}

				ordered_situation_group_participant = {

					recalculate_participant_group = situation:guelph_ghibelline
				}

		if = {
			limit = {

				exists = title:k_papal_state.holder
				title:e_hre.holder != title:k_papal_state.holder
				NOT = {
					any_situation_participant = {
						has_variable = guelph_leader
						has_variable = ghibelline_leader
					}
				}
				#If we start in 1066, the Ghibelline leader must be the HRE and the Guelph leader must be the Pope on the Concordat of Worms, otherwise we end the situation
				trigger_if = {
					limit = { gg_phase_is_pre_gg_trigger = yes }
					title:e_hre.holder = { 
						has_variable = ghibelline_leader
						#And the Pope should be the Kaiser's head of faith
						faith.religious_head = title:k_papal_state.holder
					}
					title:k_papal_state.holder = { has_variable = guelph_leader }
				}
			}
			#Carry on
		}
		else = {
			if = {
				limit = {
					OR = {
						title:e_hre.holder ?= title:k_papal_state.holder
						any_situation_participant = {
							has_variable = guelph_leader
							has_variable = ghibelline_leader
						}
					}
				}
				save_scope_value_as = {
					name = missing_leader
					value = flag:pope_is_emperor
				}
			}
			else_if = {
				limit = {
					NOT = {
						any_situation_participant = { has_variable = guelph_leader }
					}
				}
				save_scope_value_as = {
					name = missing_leader
					value = flag:guelph
				}
				gg_invalidation_missing_leader_message_effect = { MISSING_LEADER = scope:missing_leader }
			}
			else_if = {
				limit = {
					NOT = {
						any_situation_participant = { has_variable = ghibelline_leader }
					}
				}
				save_scope_value_as = {
					name = missing_leader
					value = flag:ghibelline
				}
				gg_invalidation_missing_leader_message_effect = { MISSING_LEADER = scope:missing_leader }
			}
			else_if = {
				limit = {
					NOT = { exists = title:e_hre.holder }
				}
				save_scope_value_as = {
					name = missing_leader
					value = flag:hre
				}
			}
			else_if = {
				limit = {
					game_start_date = 1066.9.15
					current_date <= 1122.9.23
					NOT = {
						title:e_hre.holder = { has_variable = ghibelline_leader }
					}
				}
				save_scope_value_as = {
					name = missing_leader
					value = flag:1066_hre_not_leader
				}
			}
			else_if = {
				limit = {
					game_start_date = 1066.9.15
					current_date <= 1122.9.23
					NOT = {
						title:k_papal_state.holder = { has_variable = guelph_leader }
					}
				}
				save_scope_value_as = {
					name = missing_leader
					value = flag:1066_pope_not_leader
				}
			}
			else = {
				save_scope_value_as = {
					name = missing_leader
					value = flag:pope

		#Change the phase?
		if = {
			limit = { situation_current_phase = agitation }
			#Ghibelline Advantage
			if = {
				limit = {
					NOT = { situation_current_phase = ghibelline_advantage }
					situation_ghibelline_group_score >= situation_score_advantage_threshold
				}
				every_situation_participant = {
					limit = { is_ai = no }
					send_interface_message = {
						type = msg_ghibelline_advantage_neutral_text
						title = ghibelline_advantage
						desc = change_situation_phase_to_ghibelline_advantage_tt

				}
				every_situation_sub_region = { change_phase = ghibelline_advantage }
			}
			#Guelph Advantage
			if = {
				limit = {
					NOT = { situation_current_phase = guelph_advantage }
					situation_guelph_group_score >= situation_score_advantage_threshold

				every_situation_participant = {
					limit = { is_ai = no }
					send_interface_message = {
						type = msg_guelph_advantage_neutral_text
						title = guelph_advantage
						desc = change_situation_phase_to_guelph_advantage_tt

					situation_ghibelline_group_score < situation_score_advantage_keep_threshold

					limit = { is_ai = no }
					send_interface_message = {
						type = msg_guelph_advantage_neutral_text
						title = ghibelline_advantage
						desc = change_situation_phase_to_guelph_advantage_tt

			}

				every_situation_participant = {
					limit = { is_ai = no }
					send_interface_message = {
						type = msg_guelph_ghibelline_agitation_neutral_text
						title = agitation
						desc = change_situation_phase_to_agitation_tt
					}
				}
				every_situation_sub_region = { change_phase = agitation }
			}
		}
		#Swap from Guelph Advantage to either Ghibelline Advantage or decay into Agitation
		if = {
			limit = { 
				situation_current_phase = guelph_advantage
				OR = {
					situation_guelph_group_score < situation_score_advantage_keep_threshold
					situation_ghibelline_group_score >= situation_score_advantage_threshold

			if = {
				limit = { situation_ghibelline_group_score >= situation_score_advantage_threshold }

					limit = { is_ai = no }
					send_interface_message = {

							is_guelph_trigger = yes

	on_yearly = {
		#Hostility only decays after a full year without a hostile catalyst
		if = {
			limit = { NOT = { has_variable = gg_recent_hostility } }
			if = {
				limit = {
					situation_current_phase = in_nomine_domini

			}
			if = {
				limit = {
					situation_current_phase = repose

				trigger_situation_catalyst = catalyst_hostility_decay
			}
		}
		if = {
			limit = { situation_current_phase = agitation }
			increment_variable_effect = {
				VAR = gg_agitation_years
				VAL = 1
			}
		}
		if = {
			limit = {
				OR = {
					situation_current_phase = ghibelline_advantage
					situation_current_phase = guelph_advantage
				}
			}
			increment_variable_effect = {
				VAR = gg_advantage_years
				VAL = 1
			}
		}
		#The leader of the side with the Advantage gains Involvement every year
		if = {
			limit = { situation_current_phase = ghibelline_advantage }
			every_situation_participant = {
				limit = { has_variable = ghibelline_leader }
				gg_involvement_interaction_effect = yes
			}
		}
		if = {
			limit = { situation_current_phase = guelph_advantage }
			every_situation_participant = {
				limit = { has_variable = guelph_leader }
				gg_involvement_interaction_effect = yes
			}
		}
		#An AI Pope may excommunicate the Emperor
		if = {
			limit = { situation_current_phase = dictatus_papae }
			random_situation_participant = {
				limit = {
					has_variable = guelph_leader
					is_ai = yes
				}
				save_scope_as = gg_excommunicator
			}
			random_situation_participant = {
				limit = { has_variable = ghibelline_leader }
				save_scope_as = gg_excommunication_target
			}
			if = {
				limit = {
					exists = scope:gg_excommunicator
					scope:gg_excommunication_target ?= {
						is_adult = yes
						faith.religious_head ?= scope:gg_excommunicator
					}
					scope:gg_excommunicator = {
						is_character_interaction_valid = {
							recipient = scope:gg_excommunication_target
							interaction = excommunicate_interaction
						}
					}
				}
				random = {
					chance = gg_pope_excommunicates_emperor_chance_value
					run_interaction = {
						interaction = excommunicate_interaction
						actor = scope:gg_excommunicator
						recipient = scope:gg_excommunication_target
						send_threshold = maybe
					}
				}
			}
		}
	}
