			}
		}
		# Transition 2: Breach & Spark -> Warring Contenders once uprisings spread in 1351+
		else_if = {
			limit = {
				situation_current_phase = phx_rtr_phase_flood_spark
				has_global_variable = phx_red_turban_unrest_begun
				current_year >= 1351
			}
			situation_top_sub_region = {
				change_phase = phx_rtr_phase_warring_factions
			}
		}
		# Transition 3: Warring Contenders -> Mandate Decided when dynasty proclaimed or Dadu+Nanjing held by Han or 1385+
		else_if = {
			limit = {
				situation_current_phase = phx_rtr_phase_warring_factions
				OR = {
					has_global_variable = phx_rtr_new_dynasty_proclaimed_global
					AND = {
						title:c_shuntian.holder.top_liege ?= { culture = culture:han }
						title:c_yingtian.holder.top_liege ?= { culture = culture:han }
					}
					current_year >= 1385
				}
			}
			situation_top_sub_region = {
				change_phase = phx_rtr_phase_resolution
			}
		}
		# Clean conclusion after era of Mandate Decided has settled
		else_if = {
			limit = {
				situation_current_phase = phx_rtr_phase_resolution
				current_year >= 1405
			}
			end_situation = yes
		}
