nanbokucho = {
    situation_group_type = minor
    is_unique = yes
    keep_full_history = yes
    map_mode = participant_groups_inside_realm
    illustration = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds"
    sub_regions = {
        nanbokucho_japan = {
            landed_titles = { e_japan }
            icon = "gfx/interface/icons/flat_icons/declare_war.dds"
        }
    }
    # Like the Guelph-Ghibelline situation, enter advantage at 70% and retain it at 60%.
    # Scores count only participating house heads, each by their own house power.
    on_monthly = {
        if = {
            limit = {
                situation_current_phase = nanbokucho_warfare
                nanbokucho_total_power > 0
                nanbokucho_north_power >= nanbokucho_advantage_threshold
            }
            every_situation_sub_region = { change_phase = nanbokucho_north_advantage }
        }
        else_if = {
            limit = {
                situation_current_phase = nanbokucho_warfare
                nanbokucho_total_power > 0
                nanbokucho_south_power >= nanbokucho_advantage_threshold
            }
            every_situation_sub_region = { change_phase = nanbokucho_south_advantage }
        }
        else_if = {
            limit = {
                situation_current_phase = nanbokucho_north_advantage
                OR = {
                    nanbokucho_total_power <= 0
                    nanbokucho_north_power < nanbokucho_advantage_keep_threshold
                    AND = { nanbokucho_total_power > 0 nanbokucho_south_power >= nanbokucho_advantage_threshold }
                }
            }
            if = {
                limit = { nanbokucho_total_power > 0 nanbokucho_south_power >= nanbokucho_advantage_threshold }
                every_situation_sub_region = { change_phase = nanbokucho_south_advantage }
            }
            else = { every_situation_sub_region = { change_phase = nanbokucho_warfare } }
        }
        else_if = {
            limit = {
                situation_current_phase = nanbokucho_south_advantage
                OR = {
                    nanbokucho_total_power <= 0
                    nanbokucho_south_power < nanbokucho_advantage_keep_threshold
                    AND = { nanbokucho_total_power > 0 nanbokucho_north_power >= nanbokucho_advantage_threshold }
                }
            }
            if = {
                limit = { nanbokucho_total_power > 0 nanbokucho_north_power >= nanbokucho_advantage_threshold }
                every_situation_sub_region = { change_phase = nanbokucho_north_advantage }
            }
            else = { every_situation_sub_region = { change_phase = nanbokucho_warfare } }
        }
    }
    participant_groups = {
        nanbokucho_north = {
            icon = "gfx/interface/icons/situations/guelph_ghibelline/ghibelline_member_group.dds"
            require_realm_in_sub_region = yes
            require_domicile_in_sub_region = yes
            map_color = { 180 35 50 }
            opposite_participant_group = { nanbokucho_south }
            opposite_participant_group_opinion = -20
            is_character_valid = {
                house ?= {
                    OR = {
                        this = house:house_ashikaga
                        AND = {
                            has_variable = nanbokucho_support_north
                            NOT = { this = house:house_daikakuji }
                        }
                        AND = {
                            NOT = { has_variable = nanbokucho_support_south }
                            exists = house_confederation
                            house_confederation ?= house:house_ashikaga.house_confederation
                        }
                    }
                }
            }
        }
        nanbokucho_south = {
            icon = "gfx/interface/icons/situations/guelph_ghibelline/guelph_member_group.dds"
            require_realm_in_sub_region = yes
            require_domicile_in_sub_region = yes
            map_color = { 245 210 90 }
            opposite_participant_group = { nanbokucho_north }
            opposite_participant_group_opinion = -20
            is_character_valid = {
                house ?= {
                    OR = {
                        this = house:house_daikakuji
                        has_variable = nanbokucho_support_south
                        AND = {
                            NOT = { has_variable = nanbokucho_support_north }
                            exists = house_confederation
                            house_confederation ?= house:house_daikakuji.house_confederation
                        }
                    }
                }
            }
        }
        nanbokucho_unaligned = {
            icon = "gfx/interface/icons/flat_icons/election_preference_over_my_candidate.dds"
            require_realm_in_sub_region = yes
            require_domicile_in_sub_region = yes
            map_color = { 130 130 130 }
            is_character_valid = { always = yes }
        }
    }
    start_phase = nanbokucho_divided
    phases = {
        nanbokucho_divided = {
            icon = "gfx/interface/icons/situations/guelph_ghibelline/guelph_ghibelline_dictatus_papae.dds"
            modifier_sets = {
                nanbokucho_rivalry_effects = {
                    icon = "gfx/interface/icons/situations/situation_modifier_character.dds"
                    nanbokucho_north = { character_modifier = { monthly_prestige_gain_mult = 0.05 } }
                    nanbokucho_south = { character_modifier = { monthly_prestige_gain_mult = 0.05 } }
                }
            }
            future_phases = {
                nanbokucho_warfare = {
                    takeover_type = points
                    takeover_points = 6
                    catalysts = { catalyst_nanbokucho_war = 2 catalyst_nanbokucho_alignment = 1 catalyst_nanbokucho_reconciliation = -2 }
                }
            }
        }
        nanbokucho_warfare = {
            icon = "gfx/interface/icons/situations/guelph_ghibelline/guelph_ghibelline_agitation.dds"
            modifier_sets = {
                nanbokucho_warfare_effects = {
                    icon = "gfx/interface/icons/situations/situation_modifier_character.dds"
                    nanbokucho_north = { county_modifier = { prosperity_growth_factor = -0.1 } }
                    nanbokucho_south = { county_modifier = { prosperity_growth_factor = -0.1 } }
                }
            }
            future_phases = {
                nanbokucho_divided = {
                    takeover_type = points
                    takeover_points = 6
                    catalysts = { catalyst_nanbokucho_reconciliation = 2 catalyst_nanbokucho_war = -2 }
                }
                nanbokucho_north_advantage = {}
                nanbokucho_south_advantage = {}
            }
        }
        nanbokucho_north_advantage = {
            icon = "gfx/interface/icons/situations/guelph_ghibelline/ghibelline_advantage.dds"
            modifier_sets = {
                nanbokucho_north_advantage_effects = {
                    icon = "gfx/interface/icons/situations/situation_modifier_character.dds"
                    nanbokucho_north = {
                        character_modifier = { monthly_prestige_gain_mult = 0.1 prosperity_growth = 0.1 }
                    }
                    nanbokucho_south = { county_modifier = { prosperity_growth_factor = -0.1 } }
                }
            }
            future_phases = { nanbokucho_warfare = {} nanbokucho_south_advantage = {} }
        }
        nanbokucho_south_advantage = {
            icon = "gfx/interface/icons/situations/guelph_ghibelline/guelph_advantage.dds"
            modifier_sets = {
                nanbokucho_south_advantage_effects = {
                    icon = "gfx/interface/icons/situations/situation_modifier_character.dds"
                    nanbokucho_south = {
                        character_modifier = { monthly_prestige_gain_mult = 0.1 prosperity_growth = 0.1 }
                    }
                    nanbokucho_north = { county_modifier = { prosperity_growth_factor = -0.1 } }
                }
            }
            future_phases = { nanbokucho_warfare = {} nanbokucho_north_advantage = {} }
        }
    }
    situation_decisions = { nanbokucho_support_north_decision nanbokucho_support_south_decision nanbokucho_seek_peace_decision }
    ending_decisions = { nanbokucho_end_division_decision }
}
