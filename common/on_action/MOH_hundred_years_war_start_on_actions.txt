on_game_start = {
	on_actions = {
		MOH_on_action_game_start
        MOH_on_struggle_rule
	}
}

MOH_on_action_game_start = {
	effect = {
        if = {
			limit = {
                game_start_date >= 1337.5.24
				game_start_date <= 1453.10.19
                NOT = { has_game_rule = Hundred_Years_War_struggle_disabled }
                NOT = { exists = struggle:Hundred_Years_War_struggle }
			}
			start_struggle = {
				struggle_type = Hundred_Years_War_struggle
				start_phase = struggle_england_france_phase_contention
			}
		}
	}
}

MOH_on_struggle_rule = {
	effect = {
		if = {
			limit = {
				has_game_rule = Hundred_Years_War_struggle_disabled
			}
			struggle:Hundred_Years_War_struggle ?= { end_struggle = yes }
		}
	}
}

Hundred_Years_War_struggle_starting_events = {
	events = {
		neutral_struggle.0001
		Hundred_Years_War_struggle.0003
	}
}

on_game_start_after_lobby = {
	on_actions = { moh_hyw_on_game_start_after_lobby }
}

moh_hyw_on_game_start_after_lobby = {
	trigger = { always = yes }
	effect = {
		moh_hyw_initialize_tours_effect = yes
	}
}
