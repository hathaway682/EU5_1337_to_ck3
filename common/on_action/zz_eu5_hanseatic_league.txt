# EU5 1337 founding cities of the Hanseatic League.
on_game_start = { on_actions = { eu5_hanseatic_league_setup } }

eu5_hanseatic_league_setup = {
	effect = {
		if = {
			limit = { has_gdp_dlc_trigger = yes }
			character:eu5_regent_lub = {
				create_confederation = {
					name = HANSEATIC_LEAGUE_NAME
					type = confederation_type:trade_league
					leading_title = title:c_lubeck
				}
			}
			scope:new_confederation ?= {
				set_variable = is_hanseatic_league
				set_variable = hanseatic_baltic_identity
				set_coa = k_hansa
				add_confederation_member_title = title:c_elbmarch
				add_confederation_member_title = title:c_wesermarch
				set_cohesion = 0
				hanseatic_apply_member_benefits_effect = { CHARACTER = character:eu5_regent_lub }
				hanseatic_apply_member_benefits_effect = { CHARACTER = character:eu5_regent_ham }
				hanseatic_apply_member_benefits_effect = { CHARACTER = character:eu5_regent_brm }
			}
		}
	}
}
