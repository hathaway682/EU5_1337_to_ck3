# Characters can use charts on their own kingdom-or-higher titles or those of their top liege.
eu5_character_has_chart_trigger = {
 OR = {
  any_held_title = { tier >= tier_kingdom has_variable = eu5_charted_$REGION$ }
  top_liege ?= { any_held_title = { tier >= tier_kingdom has_variable = eu5_charted_$REGION$ } }
 }
}

# COUNTY scope. Check the appropriate chart across the ruler and their top liege.
eu5_colonial_county_charted_trigger = {
 OR = {
  AND = { title_province = { geographical_region = eu5_continent_europe } $RULER$ = { eu5_character_has_chart_trigger = { REGION = europe } } }
  AND = { title_province = { geographical_region = eu5_continent_africa } $RULER$ = { eu5_character_has_chart_trigger = { REGION = africa } } }
  AND = { title_province = { geographical_region = eu5_continent_asia } $RULER$ = { eu5_character_has_chart_trigger = { REGION = asia } } }
  AND = { title_province = { geographical_region = eu5_subcontinent_north_america } $RULER$ = { eu5_character_has_chart_trigger = { REGION = north_america } } }
  AND = { title_province = { geographical_region = eu5_subcontinent_south_america } $RULER$ = { eu5_character_has_chart_trigger = { REGION = south_america } } }
  AND = { title_province = { geographical_region = eu5_continent_oceania } $RULER$ = { eu5_character_has_chart_trigger = { REGION = oceania } } }
  AND = { title_province = { geographical_region = eu5_continent_atlantic_ocean_continent } $RULER$ = { eu5_character_has_chart_trigger = { REGION = atlantic } } }
 }
}
# COUNTY scope. Shared by the planning UI and AI target selection.
eu5_colonial_project_target_trigger = {
 is_wilderness_county_trigger = yes
 NOT = { has_county_modifier = block_settlement_ability }
 NOT = { has_county_modifier = settling_expedition_underway }
 duchy ?= {
  NOT = {
   any_de_jure_county = {
    title_province = {
     any_great_project_in_province = {
      great_project_type = eu5_colonial_enterprise
      OR = { great_project_construction_phase_is = planned great_project_construction_phase_is = in_progress }
     }
    }
   }
  }
  OR = {
   any_de_jure_county = {
    is_wilderness_county_trigger = yes
    any_neighboring_county = { holder ?= { OR = { this = $RULER$ target_is_liege_or_above = $RULER$ } } }
   }
   AND = {
    $RULER$ = {
     culture ?= { has_innovation = innovation_pilot_books }
     #capital_province = { geographical_region = world_europe }
     any_sub_realm_county = { title_province = { is_coastal = yes } }
    }
    any_de_jure_county = {
     is_wilderness_county_trigger = yes
     title_province = { is_coastal = yes }
     eu5_colonial_county_charted_trigger = { RULER = $RULER$ }
    }
   }
  }
 }
}

# Identify the crown itself or its sovereign without requiring independence.
eu5_ai_iberian_spanish_crown_trigger = {
 OR = {
  has_title = title:e_spain
  has_title = title:k_castille
  has_title = title:k_aragon
  top_liege ?= { OR = { has_title = title:e_spain has_title = title:k_castille has_title = title:k_aragon } }
 }
}
eu5_ai_portuguese_crown_trigger = {
 OR = {
  has_title = title:k_portugal
  top_liege ?= { has_title = title:k_portugal }
 }
}
