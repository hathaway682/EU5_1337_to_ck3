# Merchant interactions

##################
# Make a Request of a Merchant
# As a Patron you can ask your Merchant Tributary for something 
##################

merchant_patron_request_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}
 # make a request, make_a_request
	category = interaction_category_diplomacy
	interface_priority = 110
	icon = icon_scheme_befriend
	desc = merchant_patron_request_interaction_desc
	greeting = positive
	notification_text = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:fund_mercenaries = yes }
				desc = MERCHANT_PATRON_REQUEST_MERCENARIES_NOTIFICATION
			}
			triggered_desc = {
				trigger = { scope:fund_maa = yes }
				desc = MERCHANT_PATRON_REQUEST_MAA_NOTIFICATION
			}
			triggered_desc = {
				trigger = { scope:fund_buildings = yes }
				desc = MERCHANT_PATRON_REQUEST_FUND_BUILDINGS_NOTIFICATION
			}
			desc = MERCHANT_PATRON_REQUEST_GOLD_NOTIFICATION
		}
	}
	common_interaction = yes
	popup_on_receive = yes
	pause_on_receive = yes

	ai_min_reply_days = 1
	ai_max_reply_days = 5

	cooldown_against_recipient = { years = 1 }

	ai_frequency_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 60
		empire = 30
		hegemony = 30
	}

	is_available = {
		any_tributary = { government_has_flag = government_is_merchant }
		NOT = { government_has_flag = government_is_merchant }
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient = {
			government_has_flag = government_is_merchant
			suzerain ?= scope:actor
			this != scope:actor
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_available_quick = {
				adult = yes
				alive = yes
				incapable = no
			}
		}
		scope:recipient = {
			is_available_quick = {
				adult = yes
				alive = yes
				imprisoned = no
				incapable = no
			}
			NOR = {
				is_at_war_with = scope:actor
				has_relation_rival = scope:actor
				house_has_feud_relation_with_trigger = { TARGET = scope:actor }
				# cannot become more indebted than strong_indebted_hook (maxed out your suzeraincard)
				has_hook_of_type = {
					target = scope:actor
					type = strong_indebted_hook
				}
			}
			custom_tooltip = {
				text = must_be_your_merchant_tributary_tt
				suzerain ?= scope:actor
			}
		}
	}

	is_highlighted = {
		scope:actor = {
			is_character_interaction_potentially_accepted = {
				recipient = scope:recipient
				interaction = merchant_patron_request_interaction
			}
		}
	}

	can_send = {
		custom_tooltip = {
			text = interactions_general.tt.must_select_one_option
			calc_true_if = {
				amount = 1
				scope:request_gold = yes
				scope:fund_maa = yes
				scope:fund_mercenaries = yes
				scope:fund_buildings = yes
			}
		}
	}

	send_options_exclusive = no

	#request gold
	send_option = {
		flag = request_gold
		localization = merchant_patron_request_interaction_request_gold
		current_description = merchant_patron_request_interaction_request_gold_desc
		starts_enabled = { always = yes }
		is_valid = {
			custom_tooltip = {
				scope:recipient.gold > scope:actor.major_gold_value
				text = merchant_patron_request_interaction.request_gold_recipient_not_enough_gold
			}
			custom_tooltip = {
				text = interactions_general.tt.only_one_request
				all_false = {
					scope:fund_maa = yes
					scope:fund_mercenaries = yes
					scope:fund_buildings = yes
				}
			}
		}
	}
	#fund men-at-arms regiment
	send_option = {
		flag = fund_maa
		localization = merchant_patron_request_interaction_fund_maa
		current_description = merchant_patron_request_interaction_fund_maa_desc
		is_valid = {
			custom_tooltip = {
				scope:recipient.gold > scope:actor.minor_gold_value
				text = merchant_patron_request_interaction.fund_maa_recipient_not_enough_gold
			}
			custom_tooltip = {
				scope:actor.maa_regiments_count < scope:actor.maa_regiments_max_count
				text = contact_list_request_interaction.request_maa_actor_too_many_regiments
			}
			custom_tooltip = {
				text = interactions_general.tt.only_one_request
				all_false = {
					scope:request_gold = yes
					scope:fund_mercenaries = yes
					scope:fund_buildings = yes
				}
			}
		}
	}
	#sponsor mercenaries
	send_option = {
		flag = fund_mercenaries
		localization = merchant_patron_request_interaction_fund_mercenaries
		current_description = merchant_patron_request_interaction_fund_maa_mercenaries_desc
		is_valid = {
			custom_tooltip = {
				scope:recipient.gold > scope:actor.minor_gold_value
				text = merchant_patron_request_interaction.fund_maa_recipient_not_enough_gold
			}
			NOT = {
				scope:actor = { has_character_modifier = merchant_patron_fund_mercenaries_modifier }
			}
			custom_tooltip = {
				text = interactions_general.tt.only_one_request
				all_false = {
					scope:request_gold = yes
					scope:fund_maa = yes
					scope:fund_buildings = yes
				}
			}
		}
	}
	# sponsor construction in patron holdings
	send_option = {
		flag = fund_buildings
		localization = merchant_patron_request_interaction_fund_buildings
		current_description = merchant_patron_request_interaction_fund_buildings_desc
		# Interaction is only avaialable for players.
		is_shown = {
			scope:recipient = { is_ai = no }
		}
		is_valid = {
			custom_tooltip = {
				text = merchant_patron_request_interaction.fund_buildings_recipient_cannot_sponsor
				scope:recipient = { government_allows = can_sponsor_construction }
			}
			custom_tooltip = {
				text = merchant_patron_request_interaction.fund_buildings_patron_not_landed
				scope:actor = { is_landed = yes }
			}
			custom_tooltip = {
				text = merchant_patron_request_interaction.fund_buildings_promise_active
				scope:recipient = {
					NOT = { has_variable = merchant_patron_fund_buildings_promise }
				}
			}
			custom_tooltip = {
				text = interactions_general.tt.only_one_request
				all_false = {
					scope:request_gold = yes
					scope:fund_maa = yes
					scope:fund_mercenaries = yes
				}
			}
		}
	}

	send_option = {
		is_shown = {
			scope:actor != scope:recipient
		}
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	cost = {
		prestige = {
			add = medium_prestige_value
			# Upgrade for tier.
			multiply = {
				value = scope:recipient.highest_held_title_tier
				# Discount barons.
				add = -1
			}
			min = 100
		}
	}

	on_accept = {
		scope:actor = {
			switch = {
				trigger = yes
				scope:fund_maa ?= {
					custom_tooltip = merchant_patron_request_interaction.fund_maa
					hidden_effect = {
						ep3_pick_random_maa_regiment_effect = {
							OWNER = scope:actor
							INCLUDE_BASIC_MAA = no
							INCLUDE_SIEGE_MAA = no
						}
						send_interface_toast = {
							type = event_toast_effect_good
							title = merchant_patron_request_interaction.fund_maa_tt
							left_icon = scope:actor
							right_icon = scope:recipient
							ep3_create_random_maa_regiment_effect = {
								OWNER = scope:actor
								SIZE = 1
							}
						}
					}
					scope:recipient = {
						remove_short_term_gold = scope:actor.minor_gold_value
						add_trait_xp = {
							trait = lifestyle_merchant
							track = creditor
							value = lifestyle_merchant_xp_gain_minor_value
						}
						merchant_creditor_capstone_prestige_effect = yes
					}
				}
				scope:fund_mercenaries ?= {
					scope:actor = {
						send_interface_toast = {
							type = event_toast_effect_good
							title = merchant_patron_request_interaction.fund_mercenaries_tt
							left_icon = scope:actor
							right_icon = scope:recipient
							add_character_modifier = {
								modifier = merchant_patron_fund_mercenaries_modifier
								years = 10
							}
						}
					}
					scope:recipient = {
						remove_short_term_gold = scope:actor.minor_gold_value
						add_trait_xp = {
							trait = lifestyle_merchant
							track = creditor
							value = lifestyle_merchant_xp_gain_minor_value
						}
						merchant_creditor_capstone_prestige_effect = yes
					}
				}
				scope:fund_buildings ?= {
					scope:actor = {
						send_interface_toast = {
							type = event_toast_effect_good
							title = merchant_patron_request_interaction.fund_buildings_patron_toast
							right_icon = scope:recipient
							custom_tooltip = merchant_patron_request_interaction.fund_buildings_patron_tt
						}
					}
					scope:recipient = {
						gdp_merchant_patron_fund_buildings_start_promise_effect = { PATRON = scope:actor }
						send_interface_toast = {
							type = event_toast_effect_good
							title = merchant_patron_request_interaction.fund_buildings_merchant_toast
							right_icon = scope:actor
							custom_tooltip = merchant_patron_request_interaction.fund_buildings_merchant_tt
						}
					}
				}
				fallback = {
					scope:actor = {
						if = {
							limit = {
								NOT = {
									has_existing_loan_from_with_instrument_trigger = {
										LENDER = scope:recipient
										INSTRUMENT = flag:none
									}
								}
							}
							set_up_new_loan_effect = {
								LENDER = scope:recipient
								AMOUNT = scope:actor.major_gold_value
								INSTRUMENT = flag:none
							}
						}
						else = {
							increase_existing_loan_effect = {
								LENDER = scope:recipient
								INSTRUMENT = flag:none
							}
						}
					}
					scope:recipient = {
						merchant_creditor_capstone_prestige_effect = yes
					}
				}
			}
			scope:recipient = {
				if = {
					limit = { is_ai = yes }
					add_opinion = {
						modifier = merchant_patron_made_request_opinion
						target = scope:actor
						opinion = -25
					}
				}
			}
			if = {
				limit = {
					NOT = { scope:fund_buildings = yes }
				}
				if = {
					limit = {
						scope:hook = yes
						has_usable_hook = scope:recipient
						scope:recipient = { is_ai = yes }
					}
					use_hook = scope:recipient
					custom_tooltip = contact_list_request_interaction.tt.hook_auto_used
				}
				else = {
					scope:recipient = {
						if = { # add indebted_hook
							limit = {
								OR = {
									scope:fund_maa = yes
									scope:fund_mercenaries = yes
									scope:fund_buildings = yes
								}
								NOT = {
									has_hook_of_type = {
										target = scope:actor
										type = strong_indebted_hook
									}
								}
								can_add_hook = {
									target = scope:actor
									type = indebted_hook
								}
							}
							add_hook = {
								target = scope:actor
								type = indebted_hook
							}
						}
					}
				}
			}
		}
		if = {
			limit = {
				NOT = { scope:fund_buildings = yes }
			}
			scope:recipient = {
				add_prestige = {
					value = scope:actor.highest_held_title_tier
					multiply = 75
				}
			}
		}
	}

	can_send_despite_rejection = no

	ai_targets = { ai_recipients = tributaries }

	auto_accept = {
		custom_description = {
			text = merchant_join_republic_hook_auto_accept_tt
			object = scope:recipient
			scope:hook = yes
			scope:actor = { has_usable_hook = scope:recipient }
			scope:recipient = { is_ai = yes }
		}
	}

	ai_accept = {
		base = 0

		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}

		modifier = {
			add = -50
			scope:recipient = {
				has_opinion_modifier = {
					modifier = merchant_patron_made_request_opinion
					target = scope:actor
				}
			}
			desc = contact_list_request_interaction.repeated_request
		}

		ai_value_modifier = {
			ai_compassion = -0.5
			ai_greed = -0.5
			min = -50
			max = 50
		}

		modifier = {
			add = 50
			desc = merchant_patron_request_interaction.desires_hook
			NOT = {
				scope:recipient = { has_hook = scope:actor }
			}
		}

		modifier = {
			add = -30
			desc = contact_list_request_interaction.request_gold.poor_recipient
			scope:request_gold ?= yes
			scope:recipient.gold < scope:actor.medium_gold_value
		}

		modifier = {
			add = 25
			desc = merchant_patron_request_interaction.request_gold.creditor
			scope:request_gold ?= yes
			scope:recipient = { has_realm_law = company_purpose_creditor }
		}

		modifier = {
			add = 25
			desc = contact_list_request_interaction.request_gold.martial
			scope:fund_maa ?= yes
			scope:recipient = { has_trait = education_martial }
		}

		modifier = {
			add = 25
			desc = contact_list_request_interaction.relevant_focus
			scope:fund_maa ?= yes
			scope:recipient = {
				OR = {
					has_focus = martial_strategy_focus
					has_focus = martial_chivalry_focus
					has_focus = martial_authority_focus
				}
			}
		}

		modifier = {
			add = 20
			desc = merchant_patron_request_interaction.fund_buildings.stewardship_focus
			scope:fund_buildings ?= yes
			scope:recipient = {
				OR = {
					has_focus = stewardship_domain_focus
					has_focus = stewardship_wealth_focus
					has_focus = stewardship_duty_focus
				}
			}
		}

		modifier = {
			add = -25
			desc = merchant_patron_request_interaction.fund_buildings.recipient_poor
			scope:fund_buildings ?= yes
			scope:recipient = { gold < medium_gold_value }
		}

		# Your level of prestige.
		## Impotent
		modifier = {
			add = -10
			desc = AI_OPINION_PRESTIGE_LEVEL_DISGRACED
			scope:actor.prestige_level = 0
		}
		## Influential
		### Would be zero, so it wouldn't appear.
		## Persuasive
		modifier = {
			add = 5
			desc = AI_OPINION_PRESTIGE_LEVEL_DISTINGUISHED
			scope:actor = {
				prestige_level = 2
				NOT = { has_perk = true_ruler_perk }
			}
		}
		## Controlling
		modifier = {
			add = 10
			desc = AI_OPINION_PRESTIGE_LEVEL_ILLUSTRIOUS
			scope:actor = {
				prestige_level = 3
				NOT = { has_perk = true_ruler_perk }
			}
		}
		## Authoritative
		modifier = {
			add = 20
			desc = AI_OPINION_PRESTIGE_LEVEL_EXALTED
			scope:actor = {
				prestige_level = 4
				NOT = { has_perk = true_ruler_perk }
			}
		}
		## Imperious
		modifier = {
			add = 30
			desc = AI_OPINION_PRESTIGE_LEVEL_LEGENDARY
			scope:actor = {
				prestige_level = 5
				NOT = { has_perk = true_ruler_perk }
			}
		}
		
		## Persuasive
		modifier = {
			add = 6
			desc = AI_OPINION_PRESTIGE_LEVEL_DISTINGUISHED
			scope:actor = {
				prestige_level = 2
				has_perk = true_ruler_perk
			}
		}
		## Controlling
		modifier = {
			add = 12
			desc = AI_OPINION_PRESTIGE_LEVEL_ILLUSTRIOUS
			scope:actor = {
				prestige_level = 3
				has_perk = true_ruler_perk
			}
		}
		## Authoritative
		modifier = {
			add = 24
			desc = AI_OPINION_PRESTIGE_LEVEL_EXALTED
			scope:actor = {
				prestige_level = 4
				has_perk = true_ruler_perk
			}
		}
		## Imperious
		modifier = {
			add = 36
			desc = AI_OPINION_PRESTIGE_LEVEL_LEGENDARY
			scope:actor = {
				prestige_level = 5
				has_perk = true_ruler_perk
			}
		}
	}

	ai_will_do = {
		base = -25
		# Set up the AI so that they actually know how to use the send_options.
		## Gold is always good.
		modifier = {
			add = 25
			scope:recipient.gold >= scope:actor.major_gold_value
			scope:actor.gold < major_gold_value
			scope:request_gold = yes
		}
		modifier = {
			add = {
				value = 10
				multiply = scope:actor.debt_level
			}
			scope:request_gold = yes
		}
		# Military aid is situational
		modifier = {
			add = 25
			scope:actor = {
				is_at_war = yes
				any_character_war = {
					is_war_leader = scope:actor
					any_war_participant = {
						is_at_war_with = scope:actor
						military_power > scope:actor.military_power
					}
				}
				gold < major_gold_value # running low
			}
			scope:fund_maa = yes
		}
		modifier = {
			add = 25
			scope:actor = {
				is_at_war = yes
				any_character_war = {
					is_war_leader = scope:actor
					any_war_participant = {
						is_at_war_with = scope:actor
						military_power > scope:actor.military_power
					}
				}
				OR = {
					gold >= major_gold_value # can afford mercs
					any_hired_mercenary = { count >= 1 } #has mercs on payroll
				}
			}
			scope:fund_mercenaries = yes
		}
		modifier = {
			add = 50
			scope:fund_buildings = yes
			scope:recipient = { gold >= medium_gold_value }
		}
	}
}

# Make a Request of your Patron
# As a Merchant you can ask your Patron for something 
##################

patron_request_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}
 # make a request, make_a_request
	category = interaction_category_vassal
	filter_tags = { market_owner }
	interface_priority = 110
	icon = icon_scheme_befriend
	desc = patron_request_interaction_desc
	notification_text = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:request_competitiveness = yes }
				desc = PATRON_REQUEST_COMPETITIVENESS_NOTIFICATION
			}
			triggered_desc = {
				trigger = { scope:request_marriage = yes }
				desc = PATRON_REQUEST_MARRIAGE_NOTIFICATION
			}
			triggered_desc = {
				trigger = { scope:request_merchant_hunt = yes }
				desc = PATRON_REQUEST_HUNT_NOTIFICATION
			}
		}
	}
	greeting = positive
	common_interaction = yes

	popup_on_receive = yes
	pause_on_receive = yes
	
	ai_min_reply_days = 1
	ai_max_reply_days = 5

	cooldown_against_recipient = { months = 8 }

	ai_frequency_by_tier = {
		barony = 0
		county = 120 # Merchant tier
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_available = {
		government_has_flag = government_is_merchant
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:actor = {
			is_tributary_of = scope:recipient
		}
		scope:actor != scope:recipient
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			is_adult = yes
			is_imprisoned = no
			NOR = {
				is_at_war_with = scope:actor
				has_relation_rival = scope:actor
				has_opinion_modifier = {
					target = scope:actor
					modifier = released_from_prison
				}
				has_opinion_modifier = {
					target = scope:actor
					modifier = eviction_ignored_opinion
				}
				house_has_feud_relation_with_trigger = { TARGET = scope:actor }
			}
		}
		custom_tooltip = {
			text = patron_request_interaction.no_requests_available
			patron_request_any_option_available_trigger = yes
		}
		scope:actor = {
			custom_tooltip = {
				text = must_be_your_merchant_patron_tt
				suzerain ?= scope:recipient
			}
		}			
	}
	is_highlighted = {
		scope:actor = {
			is_character_interaction_potentially_accepted = {
				recipient = scope:recipient
				interaction = patron_request_interaction
			}
		}
	}
	can_send = {
		scope:actor = {
			custom_description = {
				text = "character_interactions_hostile_actions_disabled_delay"
				NOT = { has_character_flag = flag_hostile_actions_disabled_delay }
			}
		}
		custom_tooltip = {
			text = interactions_general.tt.must_select_one_option
			calc_true_if = {
				amount = 1
				scope:request_competitiveness = yes
				scope:request_marriage = yes
				scope:request_merchant_hunt = yes
			}
		}
	}
	send_options_exclusive = no
	# Request Trade Privileges (Competitiveness)
	send_option = {
		flag = request_competitiveness
		starts_enabled = { always = yes	}
		is_valid = {
			patron_request_competitiveness_available_trigger = yes
			custom_tooltip = {

	ai_min_reply_days = 4

		barony = 0
		county = 240 # Merchant tier

		hegemony = 0

	is_available = {

		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }

			government_has_flag = government_is_republic

		}

				NOT = {
					any_house_member = {

									}

# Procure an Estate
# As a Merchant you ask a Patron who heads an administrative realm to settle your family within it
##################

merchant_procure_estate_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_category_diplomacy
	filter_tags = { market_owner }
	interface_priority = 30
	icon = powerful_family_shield
	greeting = positive
	desc = merchant_procure_estate_interaction_desc

	cooldown_against_recipient = { years = 10 }

	notification_text = MERCHANT_PROCURE_ESTATE_PROPOSAL

	popup_on_receive = yes
	pause_on_receive = yes

	ai_min_reply_days = 4
	ai_max_reply_days = 9

	send_options_exclusive = no

	ai_frequency_by_tier = {
		barony = 0
		county = 240 # Merchant tier
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_available = {
		government_has_flag = government_is_merchant
		house ?= { has_house_unity_parameter = unity_merchant_can_petition_for_noble_standing }
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:actor = {
			this != scope:recipient
		}
		scope:recipient = {
			is_landed = yes
			OR = {
				government_has_mechanic = administrative
				government_has_flag = government_is_japan_feudal # uses noble families but is not admin
			}
			government_allows = noble_families
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			top_liege = this
			prestige_level >= 2
			custom_tooltip = {
				text = merchant_procure_estate_company_size_tt
				company_size_value >= 2 # company_size_medium
			}
			custom_tooltip = {
				text = merchant_procure_estate_interaction_patron_tt
				OR = {
					suzerain ?= scope:recipient
					scope:recipient = { top_liege = this }
				}
			}
		}
		trigger_if = { # Only a Duke or higher can treat you right
			limit = {
				scope:recipient = { highest_held_title_tier < tier_duchy }
			}
			custom_description = {
				text = procure_estate_low_rank_tt
				always = no
			}
		}
		trigger_if = {
			limit = {
				scope:actor = { is_at_war_with = scope:recipient }
			}
			custom_description = {
				text = procure_estate_at_war_tt
				always = no
			}
		}
		scope:recipient = {
			NOT = { has_strong_hook = scope:actor }
		}
	}

	auto_accept = {
		custom_description = {
			text = merchant_procure_estate_hook_auto_accept_tt
			object = scope:recipient
			scope:hook = yes
		}
	}

	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_option = {
		flag = hook
		is_valid = {
			scope:actor = { has_usable_hook = scope:recipient }
		}
		localization = GENERIC_SPEND_A_HOOK
	}

	send_option = {
		flag = offer_gold
		localization = merchant_procure_estate_interaction_offer_gold
		current_description = merchant_procure_estate_interaction_offer_gold_desc
	}

	cost = {
		gold = {
			add = {
				value = scope:actor.merchant_join_realm_cost_value
				desc = BASE
			}
			if = {
				limit = { scope:offer_gold = yes }
				multiply = 2
			}
		}
	}

	on_accept = {
		if = {
			limit = { scope:hook = yes }
			scope:actor = { use_hook = scope:recipient }
		}
		scope:actor = {
			custom_tooltip = merchant_procure_estate_noble_family_tt
			custom_tooltip = merchant_procure_estate_company_lost_tt
			custom_tooltip = merchant_patron_arrangement_ends_tt
			if = {
				limit = {
					any_in_list = { variable = owes_me_money }
				}
				custom_tooltip = merchant_debtors_forgiven_tt
			}
			if = {
				limit = {
					any_character_artifact = { has_variable = charter_patron }
				}
				custom_tooltip = merchant_procure_estate_charter_lost_tt
			}
			hidden_effect = {
				gdp_merchant_procures_estate_effect = { GIVER = scope:recipient }
				create_character_memory = {
					type = procured_estate_memory
					participants = { benefactor = scope:recipient }
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = suzerain
	}

	ai_accept = {
		base = -25
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1
			desc = AI_OPINION_REASON
		}
		modifier = { # An outsider with no roots in the realm
			add = -30
			desc = MERCHANT_PROCURE_ESTATE_OUTSIDER_REASON
			NOR = {
				scope:actor.house ?= {
					any_house_member = {
						is_close_or_extended_family_of = scope:recipient
					}
				}
				scope:actor = {
					any_close_or_extended_family_member = {
						is_close_or_extended_family_of = scope:recipient
					}
				}
				scope:actor.dynasty ?= scope:recipient.dynasty
			}
		}
		modifier = {
			add = -20
			desc = MERCHANT_PROCURE_ESTATE_STATE_FAITH_REASON
			exists = scope:recipient.top_liege.primary_title.state_faith
			scope:actor.faith = {
				this != scope:recipient.top_liege.primary_title.state_faith
			}
		}
		modifier = {
			add = 10
			desc = MERCHANT_PROCURE_ESTATE_LOCAL_CULTURE_REASON
			scope:recipient.capital_county.culture = { this = scope:actor.culture }
		}
		modifier = {
			add = {
				value = scope:actor.dynasty.dynasty_prestige_level
				multiply = 5
			}
			exists = scope:actor.dynasty
			desc = MERCHANT_PROCURE_ESTATE_DYNASTY_REASON
		}
		modifier = { # A wealthy company is worth ennobling
			add = {
				value = scope:actor.company_size_value
				multiply = 10
			}
			desc = MERCHANT_PROCURE_ESTATE_COMPANY_REASON
		}
		modifier = {
			add = 60
			scope:actor = {
				has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = scope:recipient }
			}
			desc = MERCHANT_PROCURE_ESTATE_RELATIONSHIP_REASON
		}
		modifier = {
			add = 40
			scope:offer_gold = yes
			desc = PURCHASE_LAND_DOUBLE_GOLD_REASON
		}
		modifier = {
			add = 160
			scope:hook = yes
			desc = YOU_HAVE_HOOK_REASON
		}

		## Dealbreaker
		modifier = {
			add = -100
			desc = AI_OPINION_DEALBREAKER_ACTOR
			scope:actor = {
				is_known_dealbreaker_trigger = { KNOWER = scope:recipient }
			}
		}
	}

	ai_will_do = {
		base = -25
		ai_value_modifier = {
			ai_boldness = -0.5
			ai_greed = -0.5
		}
		modifier = { # A struggling company is better off trading its ledgers for a name
			add = 50
			trade_competitiveness < 100
		}
		modifier = { # Already dominant, no reason to give up the company
			add = -50
			company_size_value >= 3
		}
		modifier = { # Cannot afford the buy-in
			gold < merchant_join_realm_cost_value
			factor = 0
		}
	}
}

##################
# Loan Money
# As a Merchant you loan money to a ruler in need
##################

loan_money_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_category_diplomacy
	filter_tags = { market_owner }
	interface_priority = 110
	icon = loan_money_interaction
	desc = {
		desc = loan_money_interaction_desc
		triggered_desc = {
			trigger = {
				scope:recipient = {
					has_character_flag = gdp_debt_repudiator
				}
			}
			desc = loan_money_interaction_repudiator_warning_desc
		}
	}
	common_interaction = yes

	notification_text = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:open = yes }
				desc = LOAN_MONEY_INTERACTION_OPEN_NOTIFICATION
			}
			triggered_desc = {
				trigger = { scope:disguised = yes }
				desc = LOAN_MONEY_INTERACTION_DISGUISED_NOTIFICATION
			}
			desc = LOAN_MONEY_INTERACTION_FREE_NOTIFICATION
		}
	}

	popup_on_receive = yes
	pause_on_receive = yes

	ai_min_reply_days = 0
	ai_max_reply_days = 0

	cooldown_against_recipient = { years = 10 }

	ai_frequency_by_tier = {
		barony = 0
		county = 120 # Merchant tier
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_available = {
		government_has_flag = government_is_merchant
		trigger_if = {
			limit = {
				is_ai = yes
			}
			gold > 1000 # Let's make sure the AI don't spend their gold needlessly
		}
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient = {
			is_landed_or_landless_administrative = yes
		}
	}

	is_valid_showing_failures_only = {
		# Is within diplo range
		gdp_merchant_diplo_range_trigger = {
			TARGET = scope:recipient
			MERCHANT = scope:actor
		}
		# AI must always be able to clear the full debt
		trigger_if = {
			limit = {
				scope:actor = { is_ai = yes }
			}
			scope:actor.gold >= scope:recipient.gdp_loan_money_full_gold_value
		}
		# Players must be able to clear a part of the debt
		trigger_else = {
			scope:actor.gold >= scope:recipient.gdp_loan_money_gold_value
		}
		scope:recipient = {
			is_available_quick = {
				adult = yes
				imprisoned = no
			}
			custom_tooltip = {
				text = loan_money_interaction_gold_trigger_tt
				gold <= gdp_loan_money_debt_threshold_value
			}
			is_landed_or_landless_administrative = yes
			NOR = {
				is_at_war_with = scope:actor
				has_relation_rival = scope:actor
				has_opinion_modifier = {
					target = scope:actor
					modifier = released_from_prison
				}
				has_opinion_modifier = {
					target = scope:actor
					modifier = eviction_ignored_opinion
				}
				house_has_feud_relation_with_trigger = { TARGET = scope:actor }
			}
		}
	}
	is_highlighted = {
		scope:actor = {
			is_character_interaction_potentially_accepted = {
				recipient = scope:recipient
				interaction = loan_money_interaction
			}
		}
	}
	can_send = {
		scope:recipient = {
			is_landed_or_landless_administrative = yes
			gold <= gdp_loan_money_debt_threshold_value
		}
	}

	send_options_exclusive = yes
	send_option = {
		flag = open
		localization = loan_money_interaction_open_interest
		current_description = {
			first_valid = {
				triggered_desc = {
					trigger = {
						scope:actor = {
							gdp_usury_condemns_loan_trigger = {
								BORROWER = scope:recipient
							}
						}
					}
					desc = loan_money_interaction_open_interest_sin_desc
				}
				desc = loan_money_interaction_open_interest_desc
			}
		}
		starts_enabled = {
			NOT = {
				scope:actor = {
					gdp_usury_condemns_loan_trigger = {
						BORROWER = scope:recipient
					}
					NOT = {
						gdp_usury_untroubled_by_sin_trigger = yes
					}
				}
			}
		}
	}

	send_option = {
		flag = disguised
		localization = loan_money_interaction_disguised_interest
		current_description = loan_money_interaction_disguised_interest_sin_desc
		is_shown = {
			gdp_usury_either_party_condemns_loan_trigger = {
				LENDER = scope:actor
				BORROWER = scope:recipient
			}
		}
		starts_enabled = {
			gdp_usury_either_party_condemns_loan_trigger = {
				LENDER = scope:actor
				BORROWER = scope:recipient
			}
			# A lender does not start on the disguised option unless the borrower also condemns the loan.
			trigger_if = {
				limit = {
					scope:actor = { gdp_usury_untroubled_by_sin_trigger = yes }
				}
				scope:recipient = {
					gdp_usury_borrower_condemns_loan_trigger = {
						LENDER = scope:actor
					}
				}
			}
		}
	}

	send_option = {
		flag = free
		localization = loan_money_interaction_interest_free
		current_description = {
			first_valid = {
				triggered_desc = {
					trigger = {
						scope:actor = {
							rite = {
								rite_has_parameter = usury_charitable_lending
							}
						}
					}
					desc = loan_money_interaction_interest_free_pious_desc
				}
				desc = loan_money_interaction_interest_free_desc
			}
		}
	}

	on_accept = {
		if = {
			limit = { scope:free = yes }
			save_scope_value_as = { name = instrument value = flag:free }
		}
		else_if = {
			limit = { scope:disguised = yes }
			save_scope_value_as = { name = instrument value = flag:disguised }
		}
		else = {
			save_scope_value_as = { name = instrument value = flag:open }
		}
		if = {
			limit = { exists = scope:instrument }
			scope:recipient = {
				if = {
					limit = {
						NOT = {
							has_existing_loan_from_with_instrument_trigger = {
								LENDER = scope:actor
								INSTRUMENT = scope:instrument
							}
						}
					}
					set_up_new_loan_effect = {
						LENDER = scope:actor
						AMOUNT = scope:recipient.gdp_loan_principal_value
						INSTRUMENT = scope:instrument
					}
				}
				else = {
					increase_existing_loan_effect = {
						LENDER = scope:actor
						INSTRUMENT = scope:instrument
					}
				}
			}
		}
		# Info warnings
		scope:recipient = {
			if = {
				limit = { monthly_character_income < tiny_income_multiplier_value }
				custom_tooltip = loan_money_interaction_income_risk_tt
			}
			if = {
				limit = {
					OR = {
						has_trait = deceitful
						has_trait = arbitrary
						has_trait = greedy
					}
				}
				custom_tooltip = loan_money_interaction_personality_risk_tt
			}
			if = {
				limit = { debt_level >= 3 }
				custom_tooltip = loan_money_interaction_big_debt_risk_tt
			}
			if = {
				limit = {
					OR = {
						has_relation_rival = scope:actor
						has_minor_negative_or_below_house_relations_with_target_trigger = {
							FIRST = scope:actor
							SECOND = this
						}
					}
				}
				custom_tooltip = loan_money_interaction_enmity_risk_tt
			}
			if = {
				limit = {
					any_owned_story = {
						story_type = story_cycle_loan
						var:lender != scope:actor
					}
				}
				custom_tooltip = loan_money_interaction_other_loans_risk_tt
			}
			if = {
				limit = {
					OR = {
						any_targeting_faction = {
							faction_type = claimant_faction
							faction_power > 50
						}
						any_character_war = {
							primary_defender = scope:recipient
							OR = {
								using_cb = depose_war
								using_cb = claimant_faction_war
								casus_belli = {
									any_target_title = { this = scope:recipient.primary_title }
								}
							}
						}
					}
				}
				custom_tooltip = loan_money_interaction_depose_risk_tt
			}
			if = {
				limit = {
					OR = {
						any_targeting_faction = {
							faction_type = nation_fracturing_faction
							faction_power > 50
						}
						any_character_war = {
							primary_defender = scope:recipient
							OR = {
								using_cb = nation_fracturing_faction_war
								casus_belli = {
									any_target_title = {
										is_landless_type_title = no
										tier >= scope:recipient.highest_held_title_tier_minus_one
									}
								}
							}
						}
					}
				}
				custom_tooltip = loan_money_interaction_dissolution_risk_tt
			}
			if = {
				limit = {
					faith = {
						faith_hostility_level = {
							target = scope:actor.faith
							value = faith_astray_level
						}
					}
				}
				custom_tooltip = loan_money_interaction_faith_astray_risk_tt
			}
			else_if = {
				limit = {
					faith = {
						faith_hostility_level = {
							target = scope:actor.faith
							value = faith_hostile_level
						}
					}
				}
				custom_tooltip = loan_money_interaction_faith_hostile_risk_tt
			}
			else_if = {
				limit = {
					faith = {
						faith_hostility_level = {
							target = scope:actor.faith
							value = faith_evil_level
						}
					}
				}
				custom_tooltip = loan_money_interaction_faith_evil_risk_tt
			}
		}
	}

	can_send_despite_rejection = no

	ai_targets = { ai_recipients = neighboring_rulers }

	ai_accept = {
		base = -50

		opinion_modifier = {
			opinion_target = scope:actor
			multiplier = 0.25
		}

		modifier = {
			add = {
				value = scope:recipient.gold
				multiply = -1
				divide = 20
				max = 100
			}
			desc = LEND_MONEY_ACCRUED_DEBT
		}

		modifier = {
			any_owned_story = {
				story_type = story_cycle_loan
				NOT = {  var:lender ?= scope:actor }
			}
			add = {
				value = 0
				every_owned_story = {
					limit = {
						story_type = story_cycle_loan
						NOT = { var:lender ?= scope:actor }
					}
					add = -25
				}
			}
			desc = LEND_MONEY_HAS_LOANS_REASON
		}

		modifier = {
			any_owned_story = {
				story_type = story_cycle_loan
				var:lender ?= scope:actor
			}
			add = {
				value = 0
				every_owned_story = {
					limit = {
						story_type = story_cycle_loan
						var:lender ?= scope:actor
					}
					add = -50
				}
			}
			desc = LEND_MONEY_HAS_LOANS_FROM_ACTOR_REASON
		}

		### Is slightly in debt.
		modifier = {
			add = 15
			trigger = {
				scope:recipient = {
					debt_level = 1
				}
			}
			desc = LEND_MONEY_IS_IN_DEBT_REASON_1
		}
		### Is one year in debt.
		modifier = {
			add = 30
			trigger = {
				scope:recipient = {
					debt_level = 2
				}
			}
			desc = LEND_MONEY_IS_IN_DEBT_REASON_2
		}
		### Is two years in debt.
		modifier = {
			add = 45
			trigger = {
				scope:recipient = {
					debt_level = 3
				}
			}
			desc = LEND_MONEY_IS_IN_DEBT_REASON_3
		}
		### Is three years in debt.
		modifier = {
			add = 75
			trigger = {
				scope:recipient = {
					debt_level = 4
				}
			}
			desc = LEND_MONEY_IS_IN_DEBT_REASON_4
		}
		### Is four years in debt.
		modifier = {
			add = 100
			trigger = {
				scope:recipient = {
					debt_level = 5
				}
			}
			desc = LEND_MONEY_IS_IN_DEBT_REASON_5
		}
		### Is over five years in debt.
		modifier = {
			add = 200
			trigger = {
				scope:recipient = {
					debt_level >= 6
				}
			}
			desc = LEND_MONEY_IS_IN_DEBT_REASON_6
		}

		# Personality
		modifier = {
			add = -20
			has_trait = paranoid
			desc = INTERACTION_PARANOID
		}
		modifier = {
			add = -15
			has_trait = arrogant
			desc = INTERACTION_ARROGANT
		}
		modifier = {
			add = -10
			has_trait = stubborn
			desc = INTERACTION_STUBBORN
		}
		modifier = {
			add = 20
			has_trait = trusting
			desc = INTERACTION_TRUSTING
		}
		modifier = {
			add = 15
			has_trait = greedy
			desc = INTERACTION_GREEDY
		}
		modifier = {
			add = 10
			has_trait = humble
			desc = INTERACTION_HUMBLE
		}

		# Faith
		modifier = {
			add = -10
			faith = {
				faith_hostility_level = {
					target = scope:actor.faith
					value = faith_astray_level
				}
			}
			desc = AI_OPINION_FAITH_ASTRAY_ACTOR
		}
		modifier = {
			add = -25
			faith = {
				faith_hostility_level = {
					target = scope:actor.faith
					value = faith_hostile_level
				}
			}
			desc = AI_OPINION_FAITH_HOSTILE_ACTOR
		}
		modifier = {
			add = -50
			faith = {
				faith_hostility_level = {
					target = scope:actor.faith
					value = faith_evil_level
				}
			}
			desc = AI_OPINION_FAITH_EVIL_ACTOR
		}

		## Dealbreaker
		modifier = {
			add = -100
			desc = AI_OPINION_DEALBREAKER_ACTOR
			scope:actor = {
				is_known_dealbreaker_trigger = { KNOWER = scope:recipient }
			}
		}
		modifier = {
			add = 25
			trigger = {

	greeting = positive

					}
				}

									has_relation_friend = scope:prisoner

					limit = {
						any_character_struggle = {
							involvement = involved
							activate_struggle_catalyst_secondary_character_involvement_involved_trigger = {
								CATALYST = catalyst_ransom_important

							}

					}

merchant_trade_hook_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_category_diplomacy
	interface_priority = 110
	icon = hook_icon
	desc = merchant_trade_hook_interaction_desc
	notification_text = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:request_gold = yes }
				desc = TRADE_HOOK_GOLD_NOTIFICATION
			}
			triggered_desc = {
				trigger = { scope:request_prestige = yes }
				desc = TRADE_HOOK_PRESTIGE_NOTIFICATION
			}
			triggered_desc = {
				trigger = { scope:request_competitiveness = yes }
				desc = TRADE_HOOK_COMPETITIVENESS_NOTIFICATION
			}
		}
	}
	common_interaction = yes
	use_diplomatic_range = no

	popup_on_receive = yes
	pause_on_receive = yes

	ai_min_reply_days = 1
	ai_max_reply_days = 5

	cooldown_against_recipient = { years = 5 }

	ai_frequency_by_tier = {
		barony = 0
		county = 60 # Merchant tier
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_available = { 
		government_has_flag = government_is_merchant 
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:actor != scope:recipient
		scope:recipient = {
			NOT = { is_courtier_of = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { has_hook = scope:recipient }
		scope:recipient = {
			is_available_quick = {
				adult = yes
				imprisoned = no
				incapable = no
			}
			NOR = {
				is_at_war_with = scope:actor
				has_relation_rival = scope:actor
				has_opinion_modifier = {
					target = scope:actor
					modifier = released_from_prison
				}
				has_opinion_modifier = {
					target = scope:actor
					modifier = eviction_ignored_opinion
				}
				house_has_feud_relation_with_trigger = { TARGET = scope:actor }
			}
		}
	}
	is_highlighted = {
		scope:actor = {
			is_character_interaction_potentially_accepted = {
				recipient = scope:recipient
				interaction = merchant_trade_hook_interaction
			}
		}
	}
	can_send = {
		scope:actor = {
			custom_description = {
				text = "character_interactions_hostile_actions_disabled_delay"
				NOT = { has_character_flag = flag_hostile_actions_disabled_delay }
			}
		}
		custom_tooltip = {
			text = interactions_general.tt.must_select_one_option
			calc_true_if = {
				amount = 1
				scope:request_gold = yes
				scope:request_prestige = yes
				scope:request_competitiveness = yes
			}
		}
	}
	send_options_exclusive = yes
	# Request Gold
	send_option = {
		flag = request_gold
		starts_enabled = { always = yes	}
		is_valid = {
			trigger_if = {
				limit = { merchant_trade_hook_from_ransom_trigger = yes }
				scope:recipient.gold >= merchant_trade_hook_ransom_repayment_value
			}
			trigger_else_if = {
				limit = { merchant_trade_hook_from_sponsor_trigger = yes }
				scope:recipient.gold >= merchant_trade_hook_sponsor_repayment_value
			}
			trigger_else_if = {
				limit = {
					scope:actor = { has_perk = golden_obligations_perk }
				}
				scope:recipient.gold > scope:actor.medium_gold_value_twice
			}
			trigger_else = {
				scope:recipient.gold > scope:actor.medium_gold_value
			}
		}
	}
	# Request Prestige
	send_option = {
		flag = request_prestige
		is_valid = {
			scope:recipient.prestige >= scope:actor.minor_prestige_value
		}
	}
	# Request Trade Privileges (Competitiveness)
	send_option = {
		flag = request_competitiveness
		is_valid = {
			custom_tooltip = {
				text = merchant_trade_hook_competitiveness_suzerain_tt
				scope:recipient != scope:actor.suzerain # use Make a Request instead
			}
			scope:recipient = {
				highest_held_title_tier > tier_barony
				is_available_quick = {
					imprisoned = no
					incapable = no
				}
				# The Patron should always be the holder of the county where your domicile is located, but we'll do our due diligence and check anyway.
				save_temporary_scope_as = holder_check
			}
		}
	}

	should_use_extra_icon = {
		scope:actor = { has_strong_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_strong.dds"

	on_accept = {
		scope:recipient.capital_province.barony.trade_market = { save_scope_as = local_market }
		scope:actor = {
			use_hook = scope:recipient
			custom_tooltip = contact_list_request_interaction.tt.hook_auto_used
			switch = {
				trigger = yes
				scope:request_prestige ?= {
					send_interface_toast = {
						type = event_toast_effect_good
						title = contact_list_request_interaction.request_prestige_tt
						left_icon = scope:actor
						right_icon = scope:recipient
						if = {
							limit = {
								scope:actor = { has_perk = golden_obligations_perk }
							}
							scope:recipient = {
								add_prestige = {
									value = minor_prestige_gain
									multiply = -1
								}
							}
							scope:actor = {
								add_prestige = {
									value = scope:recipient.medium_prestige_gain
									multiply = 2
								}
							}
						}
						else = {
							scope:recipient = {
								add_prestige = {
									value = minor_prestige_gain
									multiply = -1
								}
							}
							scope:actor = {
								add_prestige = {
									value = scope:recipient.medium_prestige_gain
								}
							}
						}
						
						scope:recipient = {
							if = {
								limit = { is_ai = yes }
								add_opinion = {
									modifier = laamp_used_contact_opinion
									target = scope:actor
									opinion = -10
								}
							}
						}
					}
				}
				scope:request_competitiveness ?= {
					send_interface_toast = {
						type = event_toast_effect_good
						title = contact_list_request_interaction.request_competitiveness_tt
						left_icon = scope:actor
						right_icon = scope:recipient
						if = {
							limit = {
								scope:actor = { has_perk = golden_obligations_perk }
							}
							scope:local_market = {
								add_competitiveness_modifier_for_character = {
									character = scope:actor
									modifier = gdp_trade_privileges_hook_double_modifier
									years = 10
								}
							}
						}
						else = {
							scope:local_market = {
								add_competitiveness_modifier_for_character = {
									character = scope:actor
									modifier = gdp_trade_privileges_hook_modifier
									years = 10
								}
							}
						}
						scope:recipient = {
							if = {
								limit = { is_ai = yes }
								add_opinion = {
									modifier = laamp_used_contact_opinion
									target = scope:actor
									opinion = -10
								}
							}
						}
					}
				}
				fallback = {
					send_interface_toast = {
						type = event_toast_effect_good
						title = contact_list_request_interaction.request_gold_tt
						left_icon = scope:actor
						right_icon = scope:recipient
						if = {
							limit = { merchant_trade_hook_from_ransom_trigger = yes }
							scope:recipient = {
								pay_short_term_gold = {
									gold = merchant_trade_hook_ransom_repayment_value
									target = scope:actor
								}
							}
						}
						else_if = {
							limit = { merchant_trade_hook_from_sponsor_trigger = yes }
							scope:recipient = {
								pay_short_term_gold = {
									gold = merchant_trade_hook_sponsor_repayment_value
									target = scope:actor
								}
							}
						}
						else_if = {
							limit = {
								scope:actor = { has_perk = golden_obligations_perk }
							}
							scope:recipient = {
								pay_short_term_gold = {
									gold = medium_gold_value_twice
									target = scope:actor
								}
							}
						}
						else = {
							scope:recipient = {
								pay_short_term_gold = {
									gold = medium_gold_value
									target = scope:actor
								}
							}
						}
						scope:recipient = {
							if = {
								limit = { is_ai = yes }
								add_opinion = {
									modifier = laamp_used_contact_opinion
									target = scope:actor
									opinion = -10
								}
							}
						}
					}
				}
			}
			hidden_effect = {
				# One repayment per ransom hook. Strong hooks are not consumed, so only a gold request, or a hook removed by this accept, clears the record.
				if = {
					limit = {
						merchant_trade_hook_from_ransom_trigger = yes
						OR = {
							scope:request_gold ?= yes
							NOT = { scope:actor = { has_hook = scope:recipient } }
						}
					}
					scope:recipient = {
						remove_variable = merchant_ransom_hook_payer
						remove_variable = merchant_ransom_hook_gold
					}
				}
				if = {
					limit = {
						merchant_trade_hook_from_sponsor_trigger = yes
						OR = {
							scope:request_gold ?= yes
							NOT = { scope:actor = { has_hook = scope:recipient } }
						}
					}
					scope:recipient = {
						remove_variable = merchant_sponsor_hook_payer
						remove_variable = merchant_sponsor_hook_gold
					}
				}
			}
		}
	}

	on_decline = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = patron_request_interaction_decline_toast
				right_icon = scope:recipient

					if = {
						limit = { is_ai = yes }
						add_opinion = {
							target = scope:actor
							modifier = annoyed_opinion
							opinion = -10
						}
					}
				}
			}
		}
	}
	
	ai_targets = { ai_recipients = contacts }
	ai_targets = { ai_recipients = hooked_characters }
	
	ai_accept = {
		base = 50

		opinion_modifier = { # Opinion Factor
			who = scope:recipient
