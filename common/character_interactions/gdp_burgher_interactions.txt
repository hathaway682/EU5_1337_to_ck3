bestow_civic_liberties_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_category_vassal
	icon = icon_free_city_grant
	desc = bestow_civic_liberties_interaction_desc
	hidden = yes #accessed thru county view

	target_type = title
	target_source = actor
	target_filter = domain_titles
	show_all_targets = no

	auto_accept = yes
	popup_on_receive = no

	ai_frequency = 120

	is_available = {
		burgher_liege_valid_trigger = yes
		any_held_county = {
			burgher_can_receive_privilege_trigger = yes
		}
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient = scope:actor
	}

	can_be_picked_title = {
		exists = scope:target.holder

		scope:target.holder = scope:actor
		
		scope:target = { tier = tier_county }

		scope:target = {
			custom_tooltip = {
				text = bestow_civic_liberties_no_charter_left_tt
				burgher_can_receive_privilege_trigger = yes
			}
		}
	}

	options_heading = bestow_civic_liberties_options_heading
	send_options_exclusive = yes

	send_option = {
		is_shown = {
			scope:target ?= { burgher_can_demand_market_rights_trigger = yes }
		}
		flag = grant_market_rights
		localization = bestow_civic_liberties_option_market_rights
	}
	send_option = {
		is_shown = {
			scope:target ?= { burgher_can_demand_guild_rights_trigger = yes }
		}
		flag = grant_guild_rights
		localization = bestow_civic_liberties_option_guild_rights
	}
	send_option = {
		is_shown = {
			scope:target ?= { burgher_can_demand_tax_exemption_trigger = yes }
		}
		flag = grant_tax_exemption
		localization = bestow_civic_liberties_option_tax_exemption
	}
	send_option = {
		is_shown = {
			scope:target ?= { burgher_can_demand_militia_charter_trigger = yes }
		}
		flag = grant_militia_charter
		localization = bestow_civic_liberties_option_militia_charter
	}
	on_accept = {
		if = {
			limit = { scope:grant_militia_charter = yes }
			burgher_bestow_named_privilege_effect = {
				COUNTY = scope:target
				PRIVILEGE = militia_charter
			}
		}
		else_if = {
			limit = { scope:grant_tax_exemption = yes }
			burgher_bestow_named_privilege_effect = {
				COUNTY = scope:target
				PRIVILEGE = tax_exemption
			}
		}
		else_if = {
			limit = { scope:grant_guild_rights = yes }
			burgher_bestow_named_privilege_effect = {
				COUNTY = scope:target
				PRIVILEGE = guild_rights
			}
		}
		else_if = {
			limit = { scope:grant_market_rights = yes }
			burgher_bestow_named_privilege_effect = {
				COUNTY = scope:target
				PRIVILEGE = market_rights
			}
		}
		else = {
			burgher_bestow_privilege_effect = { COUNTY = scope:target }
		}

		burgher_defuse_county_factions_effect = { COUNTY = scope:target }

		scope:target = {
			add_popular_opinion = burgher_bestow_liberties_opinion_value
		}
	}

	ai_targets = {
		ai_recipients = self
	}

	ai_will_do = {
		base = 5

		modifier = {
			add = 10
			any_targeting_faction = {
				faction_is_type = burgher_autonomy_faction
			}
		}

		modifier = {
			add = 15
			scope:target ?= {
				any_title_joined_faction = {
					faction_is_type = burgher_autonomy_faction
				}
			}
		}

		modifier = {
			add = 5
			has_trait = generous
		}

		modifier = {
			factor = 0
			realm_law_authority_level_3 = yes
		}
	}
}

revoke_civic_liberties_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_category_vassal
	icon = icon_civic_liberties_revoke
	desc = revoke_civic_liberties_interaction_desc

	hidden = yes #accessed thru county view

	target_type = title
	target_source = actor
	target_filter = domain_titles
	show_all_targets = no

	auto_accept = yes
	popup_on_receive = no

	ai_frequency = 120

	is_available = {
		burgher_liege_valid_trigger = yes
		any_held_county = {
			burgher_county_has_privileges_trigger = yes
		}
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient = scope:actor
	}

	can_be_picked_title = {
		exists = scope:target.holder
		
		scope:target.holder = scope:actor

		scope:target = { tier = tier_county }

		scope:target = {
			custom_tooltip = {
				text = revoke_civic_liberties_no_privileges_tt
				burgher_county_has_privileges_trigger = yes
			}
		}
	}

	on_accept = {
		scope:target = {
			save_scope_value_as = {
				name = revoke_liberties_hit
				value = burgher_revoke_liberties_opinion_value
			}
		}

		burgher_revoke_privileges_effect = { COUNTY = scope:target }

		scope:target = {
			add_popular_opinion = scope:revoke_liberties_hit
		}

		burgher_provoke_county_factions_effect = { COUNTY = scope:target }
	}

	ai_targets = {
		ai_recipients = self
	}

	ai_will_do = {
		base = 5

		modifier = {
			add = 10
			realm_law_authority_level_3 = yes
		}

		modifier = {
			add = 5
			has_trait = arrogant
		}

		modifier = {
			factor = 0
			any_targeting_faction = {
				faction_is_type = burgher_autonomy_faction
			}
		}

		modifier = {
			factor = 0
			scope:target ?= {
				holder ?= { save_temporary_scope_as = burgher_revoke_ai_liege }
				can_title_create_faction = {
					type = burgher_autonomy_faction
					target = scope:burgher_revoke_ai_liege
				}
			}
		}
	}
}

agitate_civic_unrest_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_category_hostile
	icon = icon_scheme_incite_unrest
	desc = agitate_civic_unrest_interaction_desc
	common_interaction = yes

	interface_priority = 40

	target_type = title
	target_source = recipient
	target_filter = realm_counties
	show_all_targets = no

	auto_accept = yes
	popup_on_receive = no

	cost = {
		treasury_or_gold = scope:actor.medium_gold_value
	}

	ai_frequency = 120

	is_available = {
		OR = {
			is_landless_merchant = yes
			government_has_flag = government_is_republic
		}
	}

	scheme = gdp_civic_agitation_scheme
	send_name = START_SCHEME
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient = {
			is_landed = yes
			burgher_liege_valid_trigger = yes
			any_sub_realm_county = {
				gdp_county_can_be_agitated_trigger = { ACTOR = scope:actor }
			}
		}
		scope:actor != scope:recipient
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_available_quick = {
				adult = yes
				imprisoned = no
			}
			gdp_can_agitate_civic_unrest_trigger = yes
		}
	}

	can_be_picked_title = {
		scope:target = { tier = tier_county }
		scope:actor = {
			custom_tooltip = {
				text = agitate_civic_unrest_eligible_tt
				can_start_scheme = {
					type = gdp_civic_agitation_scheme
					target_title = scope:target
				}
			}
		}
	}

	on_accept = {
		scope:actor = {
			begin_scheme_basic_effect = {
				SCHEME_TYPE = gdp_civic_agitation_scheme
				TARGET_TYPE = target_title
				TARGET_SCOPE = scope:target
			}
		}
	}

	ai_targets = {
		ai_recipients = neighboring_rulers
	}

	ai_will_do = {
		base = 5

		modifier = {
			add = 10
			scope:target ?= {
				prosperity_level >= burgher_faction_min_prosperity_value
			}
		}

		modifier = {
			add = 10
			scope:target ?= { popular_opinion < 0 }
		}

		modifier = {
			add = 10
			is_hanseatic_league_member_trigger = yes
		}

		modifier = {
			add = 10
			gdp_is_popular_republic_head_trigger = yes
		}

		modifier = {
			add = 5
			has_trait = ambitious
		}
	}
}
