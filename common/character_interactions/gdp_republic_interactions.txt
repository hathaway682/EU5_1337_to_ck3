# Request City Charter
ask_for_permission_to_found_republic_interaction = {
	icon = icon_request_city_charter
	category = interaction_category_vassal
	desc = ask_for_permission_to_found_republic_interaction_desc
	common_interaction = no
	notification_text = ask_for_permission_to_found_republic_interaction_notification

	interface_priority = 10

	is_available = {
		gdp_basic_can_adopt_republic_government_trigger = yes
		this != liege # Must be a vassal
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:actor = {
			this != scope:recipient
			liege = scope:recipient
		}
	}

	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = {
					has_variable = has_liege_permission_to_found_republic
				}
				custom_tooltip = {
					text = granted_city_character_tooltip
					var:has_liege_permission_to_found_republic != scope:actor.liege
				}
			}
		}
	}
	
	cost = {
		gold = {
			value = 0
			if = {
				limit = {
					scope:bribe_option = yes
				}
				add = scope:recipient.major_gold_value
			}
		}
	}
	
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		flag = bribe_option
		localization = GENERIC_SPEND_GOLD
	}
	
	send_options_exclusive = no
	
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"
	
	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
			scope:recipient = { is_ai = yes }
		}
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = {
				text = ask_for_permission_to_found_republic_accept_desc
				set_variable = {
					name = has_liege_permission_to_found_republic
					value = scope:actor.liege
					years = 10
				}
			}
			custom_tooltip = ask_for_permission_to_found_republic_alert_desc
			send_interface_toast = {
				type = event_toast_text_good
				title = ask_for_permission_to_found_republic_interaction_toast
				desc = permission_to_found_republic_accepted_desc
				left_icon = scope:recipient
			}
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:recipient
				}
				use_hook = scope:recipient
			}
		}
	}

	on_decline = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_text_bad
				title = ask_for_permission_to_found_republic_interaction_toast_refused
				desc = permission_to_found_republic_decline_desc
				left_icon = scope:recipient
			}
		}
	}

#	ai_frequency_by_tier = {
#		barony = 0
#		county = 0
#		duchy = 0
#		kingdom = 0
#		empire = 0
#		hegemony = 0
#	}

	#ai_targets = { # TODO - Evaluate if we want the AI to do this and add logic if so
	#	ai_recipients = liege
	#}

	#ai_potential = {
	#	is_adult = yes
	#}

	ai_will_do = {
		base = 0
	}

	ai_accept = {
		base = -25
		
		# Opinion
		opinion_modifier = {
			opinion_target = scope:actor
			multiplier = 0.5
		}
		
		# Relation
		modifier = {
			has_relation_friend = scope:actor
			add = 15
			desc = AI_YOUR_FRIEND
		}
		modifier = {
			has_relation_lover = scope:actor
			add = 15
			desc = AI_YOUR_LOVER
		}
		modifier = {
			has_relation_rival = scope:actor
			add = -100
			desc = AI_YOUR_RIVAL
		}
		
		# House relation
		house_relation_ai_accept_modifiers = {
 			HOUSE_1_MEMBER = scope:actor
 			HOUSE_2_MEMBER = scope:recipient
 			MULTIPLIER = 0.5
 		}
		
		# Cultural reasons
		modifier = {
			culture = {
				has_cultural_tradition = tradition_hereditary_hierarchy
			}
			add = -25
			desc = culture_has_tradition_hereditary_hierarchy_reason
		}
		modifier = {
			culture = {
				has_cultural_tradition = tradition_republican_legacy
			}
			add = 15
			desc = culture_has_tradition_republican_legacy_reason
		}
		modifier = {
			culture = {
				has_cultural_tradition = tradition_parochialism
			}
			add = 10
			desc = culture_has_tradition_parochialism_reason
		}
		
		# Do the liege already have high tier republics in the realm?
		modifier = {
			any_vassal = {
				government_has_flag = government_is_republic
				highest_held_title_tier >= tier_county
			}
			add = {
				value = 0
				every_vassal = {
					limit = {
						government_has_flag = government_is_republic
						highest_held_title_tier >= tier_county
					}
					add = -5
				}
			}
			desc = has_high_tier_republican_vassals_reason
		}
		
		# Bribe option
		modifier = {
			scope:bribe_option = yes
			add = 25
			desc = GOLD_REASON
		}
		modifier = { # Greedy characters gain additional acceptance (added as a separate modifier to make it more visible)
			scope:bribe_option = yes
			has_trait = greedy
			add = 25
			desc = RECIPIENT_IS_GREEDY
		}
	}
}

	interface_priority = 58

		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 60

			#is_at_war = no
		}
	}

	auto_accept = {
		always = yes
	}

	on_auto_accept = {
	}

	on_accept = {
		scope:recipient = {
			vassal_contract_set_obligation_level = { type = republic_government_free_city level = republic_government_free_city_granted }
		}
	}

	on_decline = {
	}

	ai_targets = {
		ai_recipients = vassals
	}

	ai_will_do = {
		base = 0

		# Opinion
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = 0.25
		}

		# Personality
		modifier = {
			factor = 1.1
			has_trait = just
		}
		modifier = {
			factor = 1.1
			has_trait = trusting
		}
	}

	ai_accept = {
		base = 100
	}
}

# Negotiate Land Cession/negotiate_land_cession
gdp_republic_aquire_land_interaction = {
	icon = purchase_land_interaction
	category = interaction_category_diplomacy
	desc = gdp_republic_aquire_land_interaction_desc
	interface_priority = 50
	
	target_type = title
	target_filter = recipient_domain_titles

	cooldown_against_recipient = { years = 10 }

	greeting = positive
	notification_text = gdp_republic_aquire_land_interaction_notification
	
	ai_frequency_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 60
		empire = 40
		hegemony = 20
	}

	is_available = {
		government_has_flag = government_is_republic
		is_available_quick = {
			adult = yes
			landed = yes
		}
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient = {
			NOR = {
				target_is_liege_or_above = scope:actor
				is_liege_or_above_of = scope:actor
			}
			is_landed = yes
			highest_held_title_tier >= tier_county
			this != scope:actor
		}
	}
	
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_at_war = no
		}
		scope:recipient = {
			is_at_war = no
			custom_tooltip = {
				text = recipient_owns_coastal_or_neigboring_county_desc
				any_held_title = {
					tier = tier_county
					this != scope:recipient.capital_county
					OR = {
						any_title_to_title_neighboring_county = {
							holder = {
								OR = {
									this = scope:actor
									any_liege_or_above = { this = scope:actor }
								}
							}
						}
						is_coastal_county = yes
					}
				}
			}
		}
	}
	
	can_be_picked_title = {
		scope:target ?= {
			custom_tooltip = {
				text = county_is_not_coastal_or_neighboring_desc
				OR = {
					any_title_to_title_neighboring_county = {
						holder = {
							OR = {
								this = scope:actor
								any_liege_or_above = { this = scope:actor }
							}
						}
					}
					is_coastal_county = yes
				}
			}
			custom_tooltip = {
				text = is_recipients_capital_desc
				this != scope:recipient.capital_county
			}
		}
	}
	
	has_valid_target_showing_failures_only = {
		scope:target ?= {
			tier = tier_county
		}
	}
	
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = only_one_gold_bribe
				NOT = { scope:gold_cost_significant = yes }
			}
			scope:actor = {
				gold >= republic_aquire_land_gold_bribe_value
			}
		}
		flag = gold_cost
		localization = GENERIC_PAY_GOLD
	}
	
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = only_one_gold_bribe
				NOT = { scope:gold_cost = yes }
			}
			scope:actor = {
				gold >= {
					value = republic_aquire_land_gold_bribe_value
					multiply = 2
				}
			}
		}
		flag = gold_cost_significant
		localization = GENERIC_PAY_GOLD_SUBSTANTIAL
	}
	
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = only_one_treasury_bribe
				NOT = { scope:treasury_cost_significant = yes }
			}
			scope:actor = {
				treasury >= republic_aquire_land_treasury_bribe_value
			}
		}
		flag = treasury_cost
		localization = GENERIC_PAY_TREASURY
	}
	
	send_option = {
		is_valid = {
			custom_tooltip = {
				text = only_one_treasury_bribe
				NOT = { scope:treasury_cost = yes }
			}
			scope:actor = {
				treasury >= {
					value = republic_aquire_land_treasury_bribe_value
					multiply = 2
				}
			}
		}
		flag = treasury_cost_significant
		localization = GENERIC_PAY_TREASURY_SUBSTANTIAL
	}

	send_options_exclusive = no

	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			exists = scope:hook
			scope:hook = yes
			scope:actor = { has_strong_hook = scope:recipient }
		}
	}

	on_accept = {
		# Transfer the selected title
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = title_change
		}
		scope:target = {
			change_title_holder = {
				holder = scope:actor
				change = scope:title_change
				take_baronies = no
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		
		# Spend hook if used
		if = {
			limit = {
				scope:hook = yes
				scope:actor = { has_usable_hook = scope:recipient }
			}
			scope:actor = {
				use_hook = scope:recipient
			}
		}
		
		# Spend wealth if bribed
		if = {
			limit = {
				OR = {
					scope:gold_cost = yes
					scope:gold_cost_significant = yes
				}
			}
			scope:actor = {
				pay_short_term_gold = {
					target = scope:recipient
					gold = {
						value = republic_aquire_land_gold_bribe_value
						if = {
							limit = {
								scope:gold_cost_significant = yes
							}
							multiply = 2
						}
					}
				}
			}
		}
		if = {
			limit = {
				OR = {
					scope:treasury_cost = yes
					scope:treasury_cost_significant = yes
				}
			}
			scope:actor = {
				pay_treasury_to_gold = {
					target = scope:recipient
					value = {
						value = republic_aquire_land_treasury_bribe_value
						if = {
							limit = {
								scope:treasury_cost_significant = yes
							}
							multiply = 2
						}
					}
				}
			}
		}

		# Add a small opinion penalty
		scope:recipient = {
			if = {
				limit = {
					is_ai = yes
				}
				add_opinion = {
					target = scope:actor
					modifier = gave_up_land_opinion
					opinion = -15
				}
			}
		}
	}

	on_decline = {
	}

	ai_targets = {
		ai_recipients = neighboring_rulers
	}

	ai_will_do = {
		base = 0
	}

	ai_accept = {
		base = -50
		
		# Opinion
		opinion_modifier = {
			opinion_target = scope:actor
			multiplier = 0.25
		}
		
		# Personality
		ai_value_modifier = {
			ai_greed = {
				if = {
					limit = { ai_greed > 0 }
					value = -0.5
				}
			}
			ai_rationality = -0.5
		}
		
		# Hook
		modifier = {
			trigger = {
				scope:hook = yes
			}
			add = 40
			desc = SCHEME_WEAK_HOOK_USED
		}
		
		# Payments
		modifier = {
			trigger = {
				OR = {
					scope:gold_cost = yes
					scope:gold_cost_significant = yes
					scope:treasury_cost = yes
					scope:treasury_cost_significant = yes
				}
			}
			add = {
				value = 0
				if = {
					limit = { scope:gold_cost = yes }
					add = 20
				}
				else_if = {
					limit = { scope:gold_cost_significant = yes }
					add = 40
				}
				if = {
					limit = { scope:treasury_cost = yes }
					add = 20
				}
				else_if = {
					limit = { scope:treasury_cost_significant = yes }
					add = 40
				}
				if = {
					limit = {
						has_trait = greedy
					}
					multiply = 1.5
				}
			}
			desc = GOLD_POSITIVE_REASON
		}

		modifier = {
			add = 20
			OR = {
				scope:gold_cost = yes
				scope:gold_cost_significant = yes
				scope:treasury_cost = yes
				scope:treasury_cost_significant = yes
			}
			scope:recipient = {
				government_has_flag = government_is_theocracy
				rite ?= { rite_has_tenet = tenet_simony }
			}
			desc = SIMONY_RITE_REASON
		}
		modifier = {
			add = -50
			OR = {
				scope:gold_cost = yes
				scope:gold_cost_significant = yes
				scope:treasury_cost = yes
				scope:treasury_cost_significant = yes
			}
			scope:recipient = {
				government_has_flag = government_is_theocracy
			}
			OR = {
				scope:actor.rite ?= {
					has_tenet_status = {
						tenet = tenet_simony
						status = prohibited
					}
				}
				scope:recipient.rite ?= {
					has_tenet_status = {
						tenet = tenet_simony
						status = prohibited
					}
				}
			}
			desc = SIMONY_IS_PROHIBITED
		}

		# Does the recipient need the gold?
		modifier = {
			debt_level >= 0
			add = {
				value = 5
				multiply = debt_level
			}
			desc = PURCHASE_LAND_IS_IN_DEBT_REASON
		}
		modifier = {
			OR = {
				scope:gold_cost = yes
				scope:gold_cost_significant = yes
				scope:treasury_cost = yes
				scope:treasury_cost_significant = yes
			}
			gold >= { value = monumental_gold_value min = 300 }
			NOT = { has_trait = greedy }
			add = {
				value = 0
				if = {
					limit = { scope:gold_cost = yes }
					add = -10
				}
				else_if = {
					limit = { scope:gold_cost_significant = yes }
					add = -20
				}
				if = {
					limit = { scope:treasury_cost = yes }
					add = -10
				}
				else_if = {
					limit = { scope:treasury_cost_significant = yes }
					add = -20
				}
			}
			desc = recipient_does_not_need_gold_reason
		}
		
		# Dread
		modifier = {
			has_dread_level_towards = {
				target = scope:actor
				level = 2
			}
			add = cowed_external_reason_value
			desc = COWED_REASON
		}
		modifier = {
			has_dread_level_towards = {
				target = scope:actor
				level = 1
			}
			add = intimidated_external_reason_value
			desc = INTIMIDATED_REASON
		}
		
		# Tier difference
		modifier = { # Actor is of higher tier
			scope:actor = {
				tier_difference = {
					target = scope:recipient
					value >= 1
				}
			}
			add = {
				value = 5
				if = {
					limit = {
						tier_difference = {
							target = scope:recipient
							value = 2
						}
					}
					add = 10
				}
				if = {
					limit = {
						tier_difference = {
							target = scope:recipient
							value = 3
						}
					}
					add = 15
				}
				if = {
					limit = {
						tier_difference = {
							target = scope:recipient
							value = 4
						}
					}
					add = 20
				}
				if = {
					limit = {
						tier_difference = {
							target = scope:recipient
							value = 5
						}
					}
					add = 25
				}
			}
			desc = RANK_DIFFERENCE_REASON
		}

	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	icon = dismantle_trade_post_interaction
	category = interaction_category_vassal
	desc = dismantle_trade_post_interaction_desc
	interface_priority = 90
	common_interaction = yes
	filter_tags = { trade_post_interactions }

	target_type = district
	target_filter = trade_post
	target_source = actor

	force_notification = yes

	ai_frequency_by_tier = {
		barony = 0
		county = 60
		duchy = 60
		kingdom = 30
		empire = 30
		hegemony = 30
	}

	is_available = {
		# Base requirements
		can_have_trade_posts = yes

		# Additional AI triggers for if it should consider this at all
		trigger_if = {
			limit = { is_ai = yes }
			trade_post_count > trade_post_limit
		}
		has_gdp_dlc_trigger = yes
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient = {
			# Only use it on yourself
			this = scope:actor
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			trade_post_count >= 1
		}
	}

	is_highlighted = {
		scope:actor = {
			trade_post_count > trade_post_limit
		}
	}
	highlighted_reason = dismantle_trade_post_interaction_highlight_reason

	auto_accept = yes

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = dismantle_trade_post_interaction_notification_title
				left_icon = scope:target.district_province.county

				# Remove the target trade post
				scope:target.district_province ?= {
					remove_district_type = trade_post_location_district
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = self
	}

	# AI logic for picking district
	ai_set_target = {
		scope:actor = {
			ordered_owned_trade_post_district = {
				order_by = {
					value = 0

					# Prioritize trade posts where we don't have any trade routes
					if = {
						limit = {
							district_province.barony.trade_market = {
								NOT = {
									any_trade_route_in_market = {
										trade_route_owner = scope:actor
									}
								}
							}
						}
						add = 10
					}

					# Prioritize any trade posts not in the capital
					if = {
						limit = {
							district_province.county != scope:actor.capital_county
						}
						add = 10
					}
				}
				save_scope_as = target
			}
		}
	}

	ai_will_do = {
		base = 100
	}

	ai_accept = {
		base = 100 # Since we do this on ourselves, there's no need to refuse
	}
}

				any_owned_trade_post_district = {

			}

						scope:actor.highest_liege_matching_rule:uses_assembly = scope:recipient

						district_province != scope:recipient.capital_province
					}
				}
			}
		}
	}

	has_valid_target_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = county_is_not_in_diplomatic_range_desc
				in_diplomatic_range = scope:target.district_province.county.holder
			}
		}
	}

		scope:recipient = {
			trigger_if = {
				limit = { is_ai = no }
				scope:actor = { is_ai = no }
			}
		}
		trigger_if = {

		flag = agent_focus_success

		current_description = start_steal_trade_post_interaction.tt.agent_focus_speed

		current_description = start_steal_trade_post_interaction.tt.agent_focus_secrecy
	}

				type = event_toast_effect_neutral
				title = start_steal_trade_post_notification
				left_icon = scope:recipient
				right_icon = scope:target.district_province.county
				# Success.
				if = {
					limit = { scope:agent_focus_success ?= yes }
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = steal_trade_post
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Success.
						AGENT_1 = agent_thief
						AGENT_2 = agent_bailiff
						AGENT_3 = agent_shill
						# Speed.
						AGENT_4 = agent_overseer
						# Secrecy.
						AGENT_5 = agent_lookout
					}
				}
				# Speed.
				else_if = {
					limit = { scope:agent_focus_speed ?= yes }
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = steal_trade_post
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Speed.
						AGENT_1 = agent_overseer
						AGENT_2 = agent_infiltrator
						AGENT_3 = agent_planner
						# Success.
						AGENT_4 = agent_thief
						# Secrecy.
						AGENT_5 = agent_lookout
					}
				}
				# Secrecy.
				else_if = {
					limit = { scope:agent_focus_secrecy ?= yes }
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = steal_trade_post
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Secrecy.
						AGENT_1 = agent_lookout
						AGENT_2 = agent_justiciar
						AGENT_3 = agent_alibi
						# Success.
						AGENT_4 = agent_thief
						# Speed.
						AGENT_5 = agent_infiltrator
					}
				}
				# Balanced.
				else = {
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = steal_trade_post
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Success.
						AGENT_1 = agent_thief
						AGENT_2 = agent_bailiff
						# Speed.
						AGENT_3 = agent_infiltrator
						AGENT_4 = agent_overseer
						# Secrecy.
						AGENT_5 = agent_lookout
					}
				}
			}
			# Remember which trade post the scheme is after
			random_scheme = {
				type = steal_trade_post
				limit = {
					scheme_target_character = scope:recipient
					NOT = { has_variable = target_trade_post }
				}
				set_variable = {
					name = target_trade_post
					value = scope:trade_post_province
				}
			}
			# Scheme progress bonus from education
			education_5_scheme_progress_effect = { SKILL = intrigue }
			# Only get stressed/SP penalties if you don't have the parameter
			if = {
				limit = {
					NOT = { has_fulfillment_parameter = sp_hostile_schemes_minus_murder_no_sp_or_stress_loss }
				}
				stress_and_fulfillment_impact = {
					content = medium_stress_impact_gain
				}
			}
			# Mirrors the local ruler terms in the scheme's base_success_chance, which cannot see the chosen trade post before launch
			scope:trade_post_province ?= {
				county.holder ?= { save_scope_as = local_ruler }
			}
			if = {
				limit = { exists = scope:local_ruler }
				if = {
					limit = {
						scope:local_ruler = {
							gdp_steal_trade_post_local_ruler_favors_trigger = {

								OTHER = scope:recipient

				}

					limit = {
						scope:local_ruler = {
							gdp_steal_trade_post_local_ruler_favors_trigger = {
								FAVORED = scope:recipient
								OTHER = scope:actor
							}
						}
					}

				}
				if = {
					limit = { has_hook = scope:local_ruler }
					custom_tooltip = start_steal_trade_post_local_ruler_hooked_tt

	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_category_diplomacy
	filter_tags = { market_owner }
	interface_priority = 160
	icon = rescind_trade_monopoly
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:actor = scope:recipient
				}
				desc = rescind_trade_monopoly_interaction_self_desc
			}
			desc = rescind_trade_monopoly_interaction_desc
		}
	}

	popup_on_receive = yes
	pause_on_receive = yes

	ai_frequency_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 60
		empire = 30
		hegemony = 30
	}

	is_available = {
		# Any ruler who is the market owner of a trade market has the power to rescind monopolies
		any_held_trade_market = {
			trade_market_owner.holder = root
		}
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		scope:actor = {
			# Any market where the actor is the market owner must have a monopoly held by recipient
			any_held_trade_market = {
				trade_market_owner.holder = scope:actor
				any_trade_monopoly_for_market = {
					trigger_if = {
						limit = {
							exists = confederation_owner
						}
						scope:recipient.confederation ?= confederation_owner
					}
					trigger_else_if = {
						limit = {
							exists = title_owner
						}
						scope:recipient.primary_title ?= title_owner
					}
					trigger_else = {
						scope:recipient = character_owner
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_truce = scope:recipient }
			NOT = { is_at_war_with = scope:recipient }
		}
	}

	cost = {
		prestige = major_prestige_value
	}

	auto_accept = yes # This can never be refused

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_text_neutral
				title = rescind_trade_monopoly_interaction_msg_t
				desc = rescind_trade_monopoly_interaction_msg_desc
				left_icon = scope:recipient

				custom_tooltip = {
					text = destroy_recipients_every_monopoly_in_market_desc
					every_held_trade_market = {
						limit = {
							trade_market_owner.holder = scope:actor
						}
						every_trade_monopoly_for_market = {
							limit = {
								OR = {
									confederation_owner ?= scope:recipient.confederation
									title_owner ?= scope:recipient.primary_title
									character_owner ?= scope:recipient
								}
							}
							# Save scopes for notification event so that we can show what/who the destroyed monopolies belonged to
							if = {
								limit = { exists = confederation_owner }
								save_scope_as = confederation_monopoly
							}
							else_if = {
								limit = { exists = title_owner }
								save_scope_as = title_monopoly
							}
							else = {
								save_scope_as = character_monopoly
							}
							destroy_trade_monopoly = yes
						}
					}
				}

				# If actor is part of the same trade league, then we reduce cohesion
				if = {
					limit = {
						scope:confederation_monopoly ?= {
							scope:actor.confederation ?= confederation_owner
						}
					}
					scope:confederation_monopoly.confederation_owner = { change_cohesion = -5 }
				}
			}

			# Adjust opinion
			scope:recipient = {
				if = {
					limit = {
						is_ai = yes
						this != scope:actor
					}
					add_opinion = {
						modifier = angry_opinion
						target = scope:actor
						opinion = -40
					}
				}

				# Send a notification to recipient
				if = {
					limit = {
						is_ai = no
						this != scope:actor
					}
					trigger_event = republic_event.0030
				}
			}
		}
	}

	ai_target_quick_trigger = {
		adult = yes
	}

	ai_targets = { ai_recipients = nearby_domicile_owners }
	ai_targets = { ai_recipients = top_realm_domicile_owners }

	ai_accept = {
		base = 0
	}

	ai_will_do = {
		base = -25

		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -1
		}

		# Relations
		modifier = {
			add = -15
			has_relation_lover = scope:recipient
		}
		modifier = {
			add = -25
			has_relation_friend = scope:recipient
		}
		modifier = {
			add = 100
			has_relation_rival = scope:recipient
		}

		# Traits
		## Actor personality
		modifier = {
			add = 10
			has_trait = paranoid
		}
		modifier = {
			add = 10
			has_trait = just
		}
		modifier = {
			add = -10
			has_trait = trusting
		}
		modifier = {
			add = -10
			has_trait = content
		}
		## Dealbreaker
		modifier = {
			add = 100
			scope:recipient = {
				is_known_dealbreaker_trigger = { KNOWER = scope:actor }
			}
		}
		## Recipient personality
		modifier = {
			add = 10
			scope:recipient = {
				has_trait = deceitful
			}
		}
		modifier = {
			add = -10
			scope:recipient = {
				has_trait = gregarious
			}
		}

		# Reduce chance of removing a shared trade league monopoly
		modifier = {
			add = -25
			any_held_trade_market = {
				trade_market_owner.holder = scope:actor
				any_trade_monopoly_for_market = {
					exists = confederation_owner
					confederation_owner.confederation_type = confederation_type:trade_league
					OR = {
						scope:actor.confederation ?= confederation_owner
						scope:actor.liege.confederation ?= confederation_owner
					}
					scope:recipient.confederation ?= confederation_owner
				}
			}
		}

		# Don't be as keen to remove a monopoly held by your liege
		modifier = {
			add = -10
			liege = scope:recipient
			scope:recipient != scope:actor
		}

		# If actor has an entrepot, they are more likely to get rid of monopolies to actually get the benefits from it
		modifier = {
			add = 25
			any_held_title = {
				tier = tier_county
				county_has_district = entrepot_location_district
			}
		}
		
		# A tiny player bias, we don't want the player to lose their monopoly too easily
		modifier = {
			add = -10
			scope:recipient = { is_ai = no }
		}
	}
}
