debug_spawn_domicile_artifact_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_debug_gdp
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = merchant
	common_interaction = yes
	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		debug_only = yes
		scope:recipient = {
			government_has_flag = government_is_merchant
			exists = domicile
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			is_ai = no
			is_available_quick = {
				adult = yes
				alive = yes
				incapable = no
				imprisoned = no
			}
		}
	}

	send_option = {
		flag = dangerous_game
		localization = create_dangerous_game
	}
	send_option = {
		flag = harmless_game
		localization = create_harmless_game
	}
	send_option = {
		flag = library
		localization = create_library_artifact
	}
	send_option = {
		flag = curio
		localization = create_curio_artifact
	}

	on_accept = {
		scope:recipient = {
			switch = {
				trigger = yes
				### ARTIFACT TYPE
				scope:dangerous_game = {
					
					random_province = {
						limit = { luxury_contract_dangerous_creature_location_trigger = yes }
						save_scope_as = creature_location
					}
					luxury_contract_dangerous_creature_effect = { PROVINCE = scope:creature_location }
					create_creature_artifact_effect = {
						OWNER = scope:recipient
						CREATOR = suzerain
					}
				}
				scope:harmless_game = {
					scope:recipient = { save_scope_as = harmless_local_creature }
					random_province = {
						limit = { luxury_contract_harmless_creature_location_trigger = yes }
						save_scope_as = creature_location
					}
					luxury_contract_standard_creature_effect = { PROVINCE = scope:creature_location }
					create_creature_artifact_effect = {
						OWNER = scope:recipient
						CREATOR = suzerain
					}
				}
				scope:library = {
					create_library_artifact_effect = {
						OWNER = scope:recipient
						CREATOR = suzerain
						SUBJECT = flag:no
						TOPIC = flag:no
					}
				}
				scope:curio = {
					create_curio_artifact_effect = {
						OWNER = scope:recipient
						CREATOR = suzerain
					}
				}
			}
		}
	}
	auto_accept = yes
}

debug_grant_monopoly_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_debug_gdp
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = merchant
	common_interaction = yes
	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		debug_only = yes
		scope:recipient = {
			government_has_flag = government_is_merchant
			exists = domicile
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_tooltip = {
				text = debug_spawn_merchant_contract_interaction_route_tt
				any_trade_route = {
					any_trade_market_on_route = {
						NOT = {
							any_trade_monopoly_for_market = { character_owner ?= scope:recipient }
						}
					}
				}
			}
		}
	}

	send_option = {
		flag = exploitative
		localization = monopoly_exploitative
	}
	send_option = {
		flag = normal
		localization = monopoly_normal
	}

	on_accept = {
		scope:recipient = {
			random_trade_route = {
				limit = {
					any_trade_market_on_route = {
						NOT = {
							any_trade_monopoly_for_market = { character_owner ?= scope:recipient }
						}
					}
				}
				random_trade_market_on_route = {
					limit = {
						NOT = {
							any_trade_monopoly_for_market = { character_owner ?= scope:recipient }
						}
					}
					save_scope_as = trade_market
				}
			}
		}
		scope:trade_market ?= {
			switch = {
				trigger = yes
				scope:exploitative = {
					create_trade_monopoly = {
						type = exploitative

debug_request_trade_monopoly_for_confederation_interaction = {
	is_valid = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
	}

	category = interaction_debug_gdp
	use_diplomatic_range = no
	icon = merchant
	common_interaction = yes

	# Code handles trade good selection, type toggles and setting up the actual monopoly
	special_interaction = request_monopoly
	interface = trade_market
	target_type = trade_good_category
	target_filter = all_trade_good_categories_traded_in
	target_source = actor
	can_be_picked_trade_good_category_type = {
		# No market is selected until the player picks one in the trade market window; guard the
		# scope reference so we don't spam "undefined event target" while the list is pre-filtered.
		exists = scope:trade_market
		scope:trade_market = {
			OR = {
				any_trade_good_with_need = { has_category = scope:target }
				any_trade_good_with_surplus = { has_category = scope:target }
			}
		}
	}

	is_shown = {
		scope:actor ?= { tnh_non_wilderness_character_trigger = yes }
		scope:recipient ?= { tnh_non_wilderness_character_trigger = yes }
		debug_only = yes
		scope:actor = {
			is_confederation_member = yes
		}
		scope:recipient = {
			any_held_trade_market = {
				any_character_in_market = { this = scope:actor }
				NOT = { has_complete_trade_monopoly = yes }
			}
		}
	}

	on_accept = {
		if = {
			limit = {
				scope:is_complete_monopoly = yes
				scope:is_exploitative_monopoly = yes
			}
			custom_tooltip = request_monopoly.tt.complete_exploitative
		}
		else_if = {
			limit = { scope:is_complete_monopoly = yes }
			custom_tooltip = request_monopoly.tt.complete_normal
		}
		else_if = {
			limit = {
				exists = scope:trade_good_type
				scope:is_exploitative_monopoly = yes
			}
			custom_tooltip = request_monopoly.tt.specific_exploitative
		}
		else_if = {
			limit = { exists = scope:trade_good_type }
			custom_tooltip = request_monopoly.tt.specific_normal
		}
		hidden_effect = {
			request_monopoly_effect = {
				OWNER_PARAM = owning_confederation
				OWNER = scope:actor.confederation
			}
		}
	}
}
