# 1. Enact Harbor & Gate Quarantine (1347 Black Death)
phx_plague_enact_quarantine_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/phx_decision_black_death_quarantine.dds"
	}
	decision_group_type = realm
	desc = phx_plague_enact_quarantine_decision_desc
	selection_tooltip = phx_plague_enact_quarantine_decision_tooltip

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 24
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	is_shown = {
		is_landed = yes
		has_global_variable = phx_black_death_active
		NOT = { has_character_modifier = phx_black_death_quarantine_modifier }
	}

	is_valid = {
		OR = {
			AND = {
				exists = capital_county
				capital_county = { has_county_modifier = phx_black_death_outbreak_county_modifier }
			}
			any_sub_realm_county = { has_county_modifier = phx_black_death_outbreak_county_modifier }
			has_character_flag = phx_black_death_threatened
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
	}

	cost = {
		gold = 120
		prestige = 75
	}

	effect = {
		add_character_modifier = {
			modifier = phx_black_death_quarantine_modifier
			years = 8
		}
		if = {
			limit = {
				AND = {
					exists = capital_county
					capital_county = { has_county_modifier = phx_black_death_outbreak_county_modifier }
				}
			}
			if = {
				limit = { exists = capital_county }
				capital_county = {
					remove_county_modifier = phx_black_death_outbreak_county_modifier
					change_prosperity_progress = 25
				}
			}
		}
		if = { limit = { is_valid_for_legitimacy_change = yes } add_legitimacy = 50 }
		custom_tooltip = phx_plague_enact_quarantine_effect_tt
		hidden_effect = {
			trigger_event = phx_glb.0004
		}
	}

	ai_potential = {
		has_global_variable = phx_black_death_active
		gold >= 200
	}

	ai_will_do = {
		base = 70
	}
}
