# 1. Proclaim the Zhizheng Reforms
phx_yuan_zhizheng_reform_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/tgp_scholars.dds"
	}
	decision_group_type = dynastic_cycle
	desc = phx_yuan_zhizheng_reform_decision_desc
	selection_tooltip = phx_yuan_zhizheng_reform_decision_tooltip

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 60
	}

	is_shown = {
		has_title = title:h_china
		NOT = { has_character_modifier = phx_yuan_zhizheng_reforms_modifier }
		OR = {
			has_character_modifier = phx_yuan_bayan_regency_modifier
			has_character_modifier = phx_yuan_chao_inflation_modifier
		}
	}

	is_valid = {
		completely_controls = title:c_shuntian
		OR = {
			stewardship >= 10
			diplomacy >= 10
			prestige_level >= 2
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		top_liege = this
	}

	cost = {
		gold = 350
		prestige = 300
	}

	effect = {
		if = {
			limit = { has_character_modifier = phx_yuan_bayan_regency_modifier }
			remove_character_modifier = phx_yuan_bayan_regency_modifier
		}
		if = {
			limit = { has_character_modifier = phx_yuan_chao_inflation_modifier }
			remove_character_modifier = phx_yuan_chao_inflation_modifier
		}
		add_character_modifier = {
			modifier = phx_yuan_zhizheng_reforms_modifier
			years = 25
		}
		if = { limit = { is_valid_for_legitimacy_change = yes } add_legitimacy = 200 }
		if = {
			limit = { exists = situation:phx_red_turban_rebellion }
			situation:phx_red_turban_rebellion = {
				if = {
					limit = { OR = { situation_current_phase = phx_rtr_phase_simmering situation_current_phase = phx_rtr_phase_flood_spark situation_current_phase = phx_rtr_phase_warring_factions } }
					trigger_situation_catalyst = catalyst_phx_rtr_imperial_reform_stabilization
				}
			}
		}
		if = {
			limit = {
				has_tgp_dlc_trigger = yes
				exists = situation:dynastic_cycle
			}
			situation:dynastic_cycle = {
				if = {
					limit = { OR = { situation_current_phase = situation_dynastic_cycle_phase_instability situation_current_phase = situation_dynastic_cycle_phase_instability_conquest } }
					trigger_situation_catalyst = catalyst_event_advancement_medium_progress
				}
			}
		}
		custom_tooltip = phx_yuan_zhizheng_reform_effect_tt
		hidden_effect = {
			trigger_event = phx_yuan.0100
		}
	}

	ai_potential = {
		has_title = title:h_china
	}

	ai_will_do = {
		base = 80
	}
}
