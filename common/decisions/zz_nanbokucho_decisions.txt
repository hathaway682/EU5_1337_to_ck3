nanbokucho_support_north_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds" }
    desc = nanbokucho_support_north_decision_desc
    selection_tooltip = nanbokucho_support_north_decision_tooltip
    decision_group_type = realm
    ai_check_interval = 0
    is_shown = {
        exists = situation:nanbokucho
        is_house_head = yes
        house ?= { NOT = { this = house:house_daikakuji } }
        OR = {
            OR = { top_participant_group:nanbokucho ?= { participant_group_type = nanbokucho_south } top_participant_group:nanbokucho ?= { participant_group_type = nanbokucho_south_leader } }
            top_participant_group:nanbokucho ?= { participant_group_type = nanbokucho_unaligned }
        }
    }
    is_valid = { is_adult = yes }
    effect = {
        house = {
            remove_variable = nanbokucho_support_south
            set_variable = { name = nanbokucho_support_north value = yes }
            every_house_member = {
                limit = { is_alive = yes is_ruler = yes }
                recalculate_participant_group = situation:nanbokucho
            }
        }
        situation:nanbokucho = { trigger_situation_catalyst = catalyst_nanbokucho_alignment }
        custom_tooltip = nanbokucho_support_north_effect_tt
    }
}
