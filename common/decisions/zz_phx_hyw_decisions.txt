# 1. Assert the Claim to the French Crown
phx_hyw_press_french_claim_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_siege_warfare.dds"
	}
	decision_group_type = major
	desc = phx_hyw_press_french_claim_decision_desc
	selection_tooltip = phx_hyw_press_french_claim_decision_tooltip

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 60
		hegemony = 0
	}

	is_shown = {
		is_landed = yes
		has_title = title:k_england
		NOT = { has_title = title:k_france }
		exists = title:k_france.holder
		NOT = { has_character_flag = phx_hyw_pressed_claim_flag }
		NOT = { has_global_variable = phx_hyw_resolved }
	}

	is_valid = {
		completely_controls = title:c_middlesex
		prestige_level >= 1
		NOT = { is_allied_to = title:k_france.holder }
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		top_liege = this
	}

	cost = {
		gold = 200
		prestige = 250
	}

	effect = {
		if = {
			limit = { NOT = { has_claim_on = title:k_france } }
			add_pressed_claim = title:k_france
		}
		add_character_modifier = {
			modifier = phx_hyw_longbow_doctrine_modifier
			years = 25
		}
		spawn_army = {
			name = phx_hyw_english_longbow_host_name
			location = root.capital_province
			men_at_arms = {
				type = longbowmen
				men = 350
			}
			men_at_arms = {
				type = armored_footmen
				men = 200
			}
			levies = 600
			inheritable = no
		}
		custom_tooltip = phx_hyw_press_french_claim_effect_tt
		hidden_effect = {
			add_character_flag = phx_hyw_pressed_claim_flag
			add_character_flag = phx_hyw_free_throne_cb
			set_global_variable = phx_hyw_war_has_begun
			if = {
				limit = { NOT = { has_global_variable = phx_hyw_story_active } }
				create_story = story_phx_hundred_years_war
			}
		}
		trigger_event = phx_hyw.0100
	}

	ai_potential = {
		has_title = title:k_england
		NOT = { has_title = title:k_france }
	}

	ai_will_do = {
		base = 85
	}
}
