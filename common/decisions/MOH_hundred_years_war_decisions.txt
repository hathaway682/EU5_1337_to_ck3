			}
		}

		prestige_level >= 3
		OR = {
			has_primary_title = title:k_england
			has_primary_title = title:e_britannia

		# No other realm controls more than 20% of the region
		custom_tooltip = {
			text = struggle_england_france_ending_total_victory_decision_region_tt
			NOT = {
				struggle:Hundred_Years_War_struggle = {
					any_involved_ruler = {
						NOT = { this = root }
						is_independent_ruler = yes
						exists = primary_title
						primary_title = { is_mercenary_company = no }
						save_temporary_scope_as = england_france_region_realm
						any_county_in_region = {
							region = MOH_Hundred_Years_War_struggle_region
							percent > fp2_struggle_hostility_region_percent_decimal_value
							holder.top_liege = scope:england_france_region_realm
						}
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = { has_ach_dlc_trigger = yes }
			add_character_flag = ach_ended_struggle
		}
		if = {
			limit = { has_dlc_feature = legends }
			legend_seed_struggle_ending_effect = {
				ENDER = root
				STRUGGLE = Hundred_Years_War_struggle
			}
		}
		show_as_tooltip = {
			dynasty = { add_dynasty_prestige = 10000 }
			give_nickname = nick_the_conqueror
		}
		# Boost to culture and faith conversion
		custom_tooltip = struggle_england_france_ending_total_victory_conversion_tt
		# Choose war boost, religion boost, or political boost
		custom_tooltip = struggle_england_france_modifier_list_tt

		# Trigger a player facing event as a coda
		trigger_event = Hundred_Years_War_struggle.1002
	}

	cost = { gold = { value = 200 } }

	ai_check_interval = 120

	ai_potential = { always = yes }

	ai_will_do = { base = 100 }

}

# 局外人
struggle_england_france_ending_outsider_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
	}
	decision_group_type = struggle

	title = struggle_england_france_ending_outsider_decision
	desc = struggle_england_france_ending_outsider_decision_desc
	selection_tooltip = struggle_england_france_ending_outsider_decision_tooltip
	sort_order = 80

	is_shown = {
		is_landless_adventurer = no
		exists = struggle:Hundred_Years_War_struggle
		custom_tooltip = {
			text = struggle_england_france_uninvolved_or_external_tt
			NOR = {
				any_character_struggle = {
					is_struggle_type = Hundred_Years_War_struggle
					involvement = involved
				}
			}
		}
		# Hold land near the struggle region
		any_realm_county = {
			title_province = { geographical_region = MOH_Hundred_Years_War_struggle_region }
		}
		NOR = {
			has_title = title:k_england
			has_title = title:k_france
			has_title = title:e_britannia
			has_title = title:e_france
		}
	}

	is_valid = {
		NOT = { has_global_variable = moh_hyw_ending_resolved }
		##### General triggers #####

		# Need lots of likes
		#TODO prestige_level >= high_prestige_level
		# Struggle phase must be right
		custom_tooltip = {
			text = struggle_england_france_uninvolved_or_external_tt
			NOT = {
				capital_province = { geographical_region = MOH_Hundred_Years_War_struggle_region }
			}
		}
		custom_tooltip = {
			text = struggle_england_france_empire_tt
			AND = {
				NOT = {
					has_title = title:k_england
				}
				NOT = {
					has_title = title:k_france
				}
				NOT = {
					has_title = title:e_britannia
				}
				NOT = {
					has_title = title:e_france
				}
			}
		}

		highest_held_title_tier = tier_empire

		# Control at least 75% of the struggle region
		custom_tooltip = {
			text = struggle_france_england_ending_regional_decision_region_tt
			struggle_france_england_ending_regional_decision_percent_region_trigger = yes
		}
		calc_true_if = {
			amount >= 1
			completely_controls = title:k_england
			completely_controls = title:k_france
		}

	}


	effect = {
		if = {
			limit = { has_ach_dlc_trigger = yes }
			add_character_flag = ach_ended_struggle
		}

		show_as_tooltip = {
			dynasty = { add_dynasty_prestige = 5000 }
			if = {
				limit = {
					completely_controls = title:k_england
					exists = title:k_england
				}
				destroy_title = title:k_england
			}
			if = {
				limit = {
					completely_controls = title:k_france
					exists = title:k_france
				}
				destroy_title = title:k_france
			}
			custom_tooltip = struggle_england_france_outsider_victory_tt
		}

		trigger_event = {
			id = Hundred_Years_War_struggle.1006
		}

		custom_tooltip = MOH_Hundred_Years_War_struggle_ends_tt
	}

	cost = {}

	ai_check_interval = 120
	
	ai_potential = {}

	ai_will_do = {
		base = 100
	}
}
