	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0

		hegemony = 60

	is_shown = {
		is_landed = yes
		OR = {
			has_title = title:e_byzantium

		NOT = { has_character_flag = phx_byz_reformed_themes_flag }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:phx_byz_themes_reformed_global
			}
		}
	}

	is_valid = {
		completely_controls = title:c_constantinople
		completely_controls = title:d_thrace
		prestige_level >= 2
		OR = {
			martial >= 10
			stewardship >= 10
			has_character_modifier = phx_byz_kantakouzenos_accord_modifier
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
		top_liege = this
	}

	cost = {
		gold = 250
		prestige = 300
	}

	effect = {
		if = {
			limit = { has_character_modifier = phx_byz_1337_twilight_modifier }
			remove_character_modifier = phx_byz_1337_twilight_modifier
		}
		add_character_modifier = {
			modifier = phx_byz_theme_reform_modifier
			years = 30
		}
		add_legitimacy = 150
		if = {
			limit = {
				has_ep3_dlc_trigger = yes
				government_has_mechanic = administrative
			}
			change_influence = 250
		}
		custom_tooltip = phx_byz_reform_imperial_themes_effect_tt
		hidden_effect = {
			add_character_flag = phx_byz_reformed_themes_flag
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:phx_byz_themes_reformed_global
			}
			trigger_event = phx_byz.0100
