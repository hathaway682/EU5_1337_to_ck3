phx_byz_theme_reconquest_cb = {
	icon = de_jure_cb
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	allowed_for_character = {
		OR = {
			has_title = title:e_byzantium
			has_title = title:h_roman_empire
		}
		NOT = {
			government_has_flag = government_is_landless_adventurer
		}
	}

	allowed_for_character_display_regardless = {
		prestige_level >= 1
	}

	allowed_against_character = {
		scope:attacker.top_liege != scope:defender.top_liege
	}

	target_titles = neighbor_land_or_water
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	ai_score_mult = {
		value = 1.25
		add = common_cb_ai_score_mult_additive
		multiply = aggressive_cb_ai_score_mult
		multiply = common_cb_ai_score_mult
	}

	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			OR = {
				target_is_de_jure_liege_or_above = title:k_nikaea
				target_is_de_jure_liege_or_above = title:k_rum
				target_is_de_jure_liege_or_above = title:k_hellas
				target_is_de_jure_liege_or_above = title:k_epirus
				target_is_de_jure_liege_or_above = title:k_thessalonika
				target_is_de_jure_liege_or_above = title:k_anatolia
				target_is_de_jure_liege_or_above = title:k_trebizond
				this = title:d_epirus
			}
		}
		ai_aggressive_cb_constraints = yes
	}

	should_invalidate = {
		war_should_invalidate_defender_lost_target_lands_trigger = yes
	}

	on_invalidated_desc = phx_byz_theme_reconquest_cb_invalidated_desc
	on_invalidated = {}

	cost = {
		piety = {
			add = 50
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			add = 250
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = phx_byz_theme_reconquest_cb_victory_desc_attacker
			}
			desc = phx_byz_theme_reconquest_cb_victory_desc
		}
	}

	on_victory = {
		on_victory_aggressive_war_effect = yes
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}

		every_in_list = {
			list = target_titles
			custom_tooltip = CONQUEST_CB_TITLE

			conquest_cb_title_transfer = {
				RELIGIOUS_WAR = no
			}
		}

		every_in_list = {
			list = vassals_taken
			change_liege = {
				liege = scope:attacker
				change = scope:change
			}
		}

		every_in_list = {
			list = titles_taken
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}

		resolve_title_and_vassal_change = scope:change

		scope:attacker = {
			add_legitimacy = 50
			add_prestige_experience = medium_prestige_value
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_white_peace_desc = {
		desc = phx_byz_theme_reconquest_cb_white_peace_desc
	}

	on_white_peace = {
		on_white_peace_effect = yes
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_defeat_desc = {
		desc = phx_byz_theme_reconquest_cb_defeat_desc
	}

	on_defeat = {
		on_defeat_aggressive_war_effect = yes
		scope:defender = {
			accolade_defender_war_end_glory_gain_med_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = major_prestige_value
			}
		}
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 3
			}
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "PHX_BYZ_THEME_RECONQUEST_WAR_NAME"
	war_name_base = "PHX_BYZ_THEME_RECONQUEST_WAR_NAME_BASE"
	cb_name = "PHX_BYZ_THEME_RECONQUEST_CB_NAME"

	interface_priority = 85
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
