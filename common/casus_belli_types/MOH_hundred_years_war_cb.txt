moh_hundred_years_war_border_raid = {
	group = struggle
	icon = struggle
	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	allow_hostages = no
	interface_priority = 79

	target_titles = neighbor_land
	target_title_tier = county
	target_de_jure_regions_above = no
	use_de_jure_wargoal_only = yes

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150
	defender_ticking_warscore_delay = { months = 3 }
	defender_ticking_warscore = 0.15

	allowed_for_character = {
		is_landed = yes
		any_character_struggle = {
			involvement = involved
			is_struggle_type = Hundred_Years_War_struggle
			has_struggle_phase_parameter = moh_unlocks_hundred_years_war_border_raid
		}
	}

	allowed_against_character = {
		is_landed = yes
		any_character_struggle = {
			involvement = involved
			is_struggle_type = Hundred_Years_War_struggle
			has_struggle_phase_parameter = moh_unlocks_hundred_years_war_border_raid
		}
		top_liege != scope:attacker.top_liege
		OR = {
			character_is_land_realm_neighbor = scope:attacker
			AND = {
				exists = scope:attacker.liege
				character_is_land_realm_neighbor = scope:attacker.liege
			}
		}
	}

	valid_to_start = {
		exists = scope:target
		scope:target.title_province = {
			geographical_region = MOH_Hundred_Years_War_struggle_region
			NOT = { has_province_modifier = recently_looted_modifier }
		}
	}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				holder ?= { target_is_same_character_or_above = scope:defender }
			}
		}
	}
	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {}
	transfer_behavior = invalidate
	on_primary_attacker_death = invalidate
	on_primary_defender_death = inherit
	attacker_allies_inherit = no
	defender_allies_inherit = no

	cost = {
		prestige = {
			value = minor_prestige_value
			multiply = common_cb_prestige_cost_multiplier
		}
	}
	ai_score_mult = {
		value = 0.5
		multiply = frankokratia_leader_protection_value
	}

	on_declaration = { on_declared_war = yes }

	on_victory_desc = { desc = moh_hundred_years_war_border_raid_victory_desc }
	on_victory = {
		every_in_list = {
			list = target_titles
			scope:attacker = {
				add_gold = {
					value = prev.prosperity_level
					multiply = 5
				}
				add_prestige_experience = miniscule_prestige_value
			}
			change_county_control = -25
			change_prosperity_level = -1
			title_province = {
				add_province_modifier = {
					modifier = recently_looted_modifier
					years = 5
				}
			}
		}
		add_truce_attacker_victory_effect = yes
		hidden_effect = {
			struggle:Hundred_Years_War_struggle ?= {
				if = {
					limit = { phase_has_catalyst = moh_hyw_catalyst_border_raid }
					activate_struggle_catalyst = {
						catalyst = moh_hyw_catalyst_border_raid
						character = scope:attacker
					}
				}
			}
		}
	}

	on_white_peace_desc = { desc = moh_hundred_years_war_border_raid_white_peace_desc }
	on_white_peace = {
		scope:attacker = { add_prestige = -35 }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = { desc = moh_hundred_years_war_border_raid_defeat_desc }
	on_defeat = {
		scope:attacker = {
			pay_short_term_gold_reparations_effect = { GOLD_VALUE = 1 }
			add_prestige = -150
		}
		scope:defender = {
			add_prestige_war_defender_effect = { PRESTIGE_VALUE = medium_prestige_value }
		}
		add_truce_attacker_defeat_effect = yes
	}

	war_name = "MOH_HUNDRED_YEARS_WAR_BORDER_RAID_WAR_NAME"
	war_name_base = "MOH_HUNDRED_YEARS_WAR_BORDER_RAID_WAR_NAME_BASE"
	cb_name = "MOH_HUNDRED_YEARS_WAR_BORDER_RAID_CB_NAME"
}
