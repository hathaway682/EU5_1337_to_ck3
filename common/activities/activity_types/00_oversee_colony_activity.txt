activity_oversee_colony = {



	is_shown = {
		tnh_colonization_enabled_trigger = yes
		is_landed = yes
		is_adult = yes
		is_colonist_trigger = no

		any_held_title = {
			tier = tier_county
			is_colony_county_trigger = yes
		}
	}

	can_always_plan = no

	can_start_showing_failures_only = {
		is_available = yes

		custom_tooltip = {
			text = colonize_not_while_at_war_tt
			is_at_war = no
		}
	}

	is_valid = {
		scope:host = {
			is_alive = yes
			is_imprisoned = no
			is_incapable = no
			is_landed = yes
		}
	}











	is_location_valid = {
		custom_tooltip = {
			text = colonize_oversee_must_hold_tt
			county ?= {
				holder = scope:host
				is_colony_county_trigger = yes
			}
		}
	}

	on_invalidated = {
		invalidation_imprisoned_effect = yes
		invalidation_incapable_effect = yes
		invalidation_unlanded_effect = yes
	}




	on_enter_active_state = {
		scope:host = { colony_refresh_presence_effect = yes }
	}




	on_complete = {
		scope:host = { colony_refresh_presence_effect = yes }
	}

	province_filter = domain
	ai_province_filter = domain

	max_province_icons = 3
	allow_zero_guest_invites = yes
	open_invite = no

	province_description = {
		desc = activity_oversee_colony_province_desc
	}
















	phases = {
		phase_overseeing = {
			is_predefined = yes
			location_source = pickable



			on_phase_active = {
				if = {
					limit = { this = scope:host }
					scope:activity = { progress_activity_phase_after = { years = 50 } }
				}
			}
		}
	}









	host_intents = {
		default = oversee_colony_intent
		intents = { oversee_colony_intent }
	}




	guest_intents = {
		default = oversee_colony_intent
		intents = { oversee_colony_intent }
	}







	max_guests = { value = 0 }













	ai_will_do = { value = 0 }
	ai_check_interval = 0














	header_background = "gfx/interface/illustrations/activity_header_backgrounds/activity_hunt.dds"
	background = {
		trigger = { always = yes }
		texture = "gfx/interface/illustrations/event_scenes/forest.dds"
		environment = "environment_event_forest"
		ambience = "event:/SFX/Events/Backgrounds/deciduous_forest_day"
	}

	window_characters = {
		host = {
			camera = camera_body

			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				animation = personality_honorable
			}
		}

		travel_host = {
			camera = camera_event_horse_left

			effect = {
				if = {
					limit = { scope:host.location != scope:activity.activity_location }
					scope:host = { add_to_list = characters }
				}
			}

			scripted_animation = {
				animation = jockey_walk
			}
		}
	}
}
