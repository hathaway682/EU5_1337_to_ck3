# Each participating house head contributes its house power score once.
nanbokucho_north_power = {
    value = 0
    situation:nanbokucho ?= {
        situation_participant_group:nanbokucho_north ?= {
            every_situation_group_participant = {
                limit = { is_house_head = yes exists = house }
                add = house.house_power_score
            }
        }
    }
}

nanbokucho_south_power = {
    value = 0
    situation:nanbokucho ?= {
        situation_participant_group:nanbokucho_south ?= {
            every_situation_group_participant = {
                limit = { is_house_head = yes exists = house }
                add = house.house_power_score
            }
        }
    }
}

nanbokucho_total_power = { value = nanbokucho_north_power add = nanbokucho_south_power }
nanbokucho_advantage_threshold = { value = nanbokucho_total_power multiply = 0.7 }
nanbokucho_advantage_keep_threshold = { value = nanbokucho_total_power multiply = 0.6 }
nanbokucho_north_share = {
    value = 0
    if = { limit = { nanbokucho_total_power > 0 } value = nanbokucho_north_power divide = nanbokucho_total_power multiply = 100 }
}
nanbokucho_south_share = {
    value = 0
    if = { limit = { nanbokucho_total_power > 0 } value = nanbokucho_south_power divide = nanbokucho_total_power multiply = 100 }
}
