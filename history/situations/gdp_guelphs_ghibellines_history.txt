#In Nomine Domini
1059.4.13 = {
	effect = {
		if = {
			limit = { has_gdp_dlc_trigger = yes }
			start_situation = {
				type = guelph_ghibelline
				start_phase = in_nomine_domini
			}
			situation:guelph_ghibelline = {
				#This should never happen (we should never return to this phase), but just making sure
				if = {
					limit = {
						NOT = { has_variable = recorded_situation_start }
					}
					record_situation_special_event = {
						key = situation_gg_start_situation_phase_entry
						phase = in_nomine_domini
					}
					set_variable = recorded_situation_start
				}
			}
		}		
	}
}

		if = {
			limit = { 
				has_gdp_dlc_trigger = yes

				if = {
					limit = { situation_current_phase = agitation }

						key = gg_history_key_alexander_iii_election

							alexander_iii = scope:char_alexander_iii
						}

				}

						phase = repose

							alexander_iii = scope:char_alexander_iii

	}
