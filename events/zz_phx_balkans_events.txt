# 0001 — Opening 1337: Banner of Osman's Heir
phx_bal.0001 = {
	type = character_event
	title = phx_bal.0001.t
	desc = phx_bal.0001.desc
	theme = war
	override_background = { reference = army_camp }
	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		trigger = { exists = scope:ottoman_prince }
		character = scope:ottoman_prince
		animation = marshal
	}

	immediate = {
		if = {
			limit = {
				character:tur_suleyman ?= {
					is_alive = yes
					NOT = { this = root }
				}
			}
			character:tur_suleyman = { save_scope_as = ottoman_prince }
		}
		else_if = {
			limit = {
				character:tur_alaeddin_pasha ?= {
					is_alive = yes
					NOT = { this = root }
				}
			}
			character:tur_alaeddin_pasha = { save_scope_as = ottoman_prince }
		}
	}

	# Option A: Rally the Ghazi riders of Bithynia for the western frontier
	option = {
		name = phx_bal.0001.a
		add_character_modifier = {
			modifier = phx_ott_ghazi_zeal_modifier
			years = 15
		}
		add_prestige = 120
		ai_chance = { base = 65 }
	}

	# Option B: Build up Bursa's bazaars, coinage, and pious endowments
	option = {
		name = phx_bal.0001.b
		add_gold = 125
		if = {
			limit = { completely_controls = title:c_hudavendigar }
			title:c_hudavendigar = {
				change_prosperity_progress = 60
			}
		}
		add_piety = 75
		ai_chance = { base = 35 }
	}
}

# 0005 — The Shattered Walls of Gallipoli (1354 Earthquake Crossing)
phx_bal.0005 = {
	type = character_event
	title = phx_bal.0005.t
	desc = phx_bal.0005.desc
	theme = disaster
	override_background = { reference = docks }
	left_portrait = {
		character = root
		animation = war_attacker
	}
	right_portrait = {
		trigger = { exists = scope:byzantine_emperor }
		character = scope:byzantine_emperor
		animation = shock
	}

	immediate = {
		if = {
			limit = {
				exists = title:e_byzantium.holder
				title:e_byzantium.holder != root
			}
			title:e_byzantium.holder = { save_scope_as = byzantine_emperor }
		}
	}

	# Option A: Ferry Ghazi regiments across the Hellespont and claim Gallipoli
	option = {
		name = phx_bal.0005.a
		add_character_flag = phx_ott_straits_crossed
		add_prestige = 150
		add_piety = 100
		if = {
			limit = { completely_controls = title:c_dardanelles }
			title:c_dardanelles = {
				add_county_modifier = {
					modifier = phx_ott_gallipoli_bridgehead_modifier
					years = 25
				}
			}
		}
		else = {
			add_unpressed_claim = title:c_dardanelles
			add_unpressed_claim = title:d_thrace
		}
		spawn_army = {
			name = phx_ott_gallipoli_raiders_name
			location = root.capital_province
			men_at_arms = {
				type = horse_archers
				men = 250
			}
			men_at_arms = {
				type = light_horsemen
				men = 150
			}
			inheritable = no
		}
		custom_tooltip = phx_ott_gallipoli_seize_tt
		ai_chance = { base = 85 }
	}

	# Option B: Loot the ruined Thracian coast and return with silver
	option = {
		name = phx_bal.0005.b
		add_gold = 200
		add_prestige = 75
		ai_chance = { base = 15 }
	}
}
