nwt_voyage.30 = {
	type = character_event
	title = nwt_voyage.30.t
	desc = { desc = nwt_voyage.30.desc desc = aod_vs_status }
	theme = realm
	override_background = { reference = ocean }
	left_portrait = {
		character = scope:nwt_captain
		animation = worry
	}
	trigger = { exists = scope:nwt_captain }

	option = {
		name = nwt_voyage.30.a
		random_list = {
			60 = {
				modifier = { add = { value = scope:nwt_captain.martial multiply = 3 } }
				desc = nwt_voyage.30.a.good
				scope:nwt_captain = { add_prestige = 50 }
				aod_vs_morale_add_10 = yes
			}
			25 = {
				desc = nwt_voyage.30.a.hurt
				scope:nwt_captain = { add_trait = wounded_1 }
				aod_vs_crew_sub_15 = yes
			}
			12 = {
				modifier = { add = { value = scope:nwt_captain.martial multiply = -0.5 } }
				desc = nwt_voyage.30.a.lost
				nwt_voyage_lost_effect = { REASON = death_sea_storm }
			}
		}
		ai_chance = { base = 50 }
	}
	option = {
		name = nwt_voyage.30.b
		custom_tooltip = eu5_voyage_delay_45_tt
		hidden_effect = { scope:eu5_voyage_title = { change_variable = { name = eu5_voyage_delay add = 45 } } }
		aod_vs_supplies_sub_10 = yes
		ai_chance = { base = 50 }
	}
}
