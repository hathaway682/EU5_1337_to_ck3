# 0003 — The Pestilence Enters the Gates (Core Black Death Choice)
phx_glb.0003 = {
	type = character_event
	title = phx_glb.0003.t
	desc = phx_glb.0003.desc
	theme = plague
	override_background = { reference = ep3_city_gate }
	left_portrait = {
		character = root
		animation = grief
	}

	immediate = {
		if = {
			limit = {
				NOT = { has_character_modifier = phx_black_death_quarantine_modifier }
				AND = {
					exists = capital_county
					capital_county = {
						NOT = { has_county_modifier = phx_black_death_outbreak_county_modifier }
					}
				}
			}
			if = {
				limit = { exists = capital_county }
				capital_county = {
					add_county_modifier = {
						modifier = phx_black_death_outbreak_county_modifier
						years = 5
					}
				}
			}
		}
	}

	# Option A: Seal the gates and enforce strict municipal quarantine
	option = {
		name = phx_glb.0003.a
		remove_short_term_gold = 100
		add_character_modifier = {
			modifier = phx_black_death_quarantine_modifier
			years = 6
		}
		if = {
			limit = {
				AND = {
					exists = capital_county
					capital_county = { has_county_modifier = phx_black_death_outbreak_county_modifier }
				}
			}
			if = {
				limit = { exists = capital_county }
				capital_county = {
					remove_county_modifier = phx_black_death_outbreak_county_modifier
				}
			}
		}
		hidden_effect = {
			trigger_event = { id = phx_glb.0007 months = 6 }
		}
		ai_chance = { base = 45 }
	}

	# Option B: Keep the chapels and hospitals open with royal alms
	option = {
		name = phx_glb.0003.b
		remove_short_term_gold = 80
		add_piety = 150
		if = { limit = { is_valid_for_legitimacy_change = yes } add_legitimacy = 75 }
		hidden_effect = {
			trigger_event = { id = phx_glb.0006 months = 5 }
		}
		ai_chance = { base = 35 }
	}

	# Option C: Withdraw the court to secluded highland estates
	option = {
		name = phx_glb.0003.c
		add_prestige = -100
		add_stress = -20
		add_character_modifier = {
			modifier = phx_black_death_quarantine_modifier
			years = 4
		}
		ai_chance = { base = 20 }
	}
}

# 0020 — Hidden Spawner: Emergence of Timur in Transoxiana (c_samarkand)
phx_glb.0020 = {
	hidden = yes

	immediate = {
		title:c_samarkand = { save_scope_as = samarkand_county }
		title:c_samarkand.title_province = { save_scope_as = samarkand_province }

		# Reuse historical 1336-born Timur if alive; otherwise dynamically create_character
		if = {
			limit = {
				character:chg_timur_lame_borjigin ?= {
					is_alive = yes
				}
			}
			character:chg_timur_lame_borjigin = {
				save_scope_as = timur_ruler
			}
		}
		else = {
			create_character = {
				name = "Timur"
				gender = male
				age = 32
				location = scope:samarkand_province
				culture = culture:kuzhu_moghol
				faith = faith:sunni
				dynasty = generate
				martial = 22
				diplomacy = 14
				stewardship = 15
				intrigue = 16
				learning = 13
				prowess = 18
				trait = education_martial_4
				trait = ambitious
				trait = brave
				trait = wrathful
				trait = strategist
				trait = conqueror
				trait = clubfooted
				save_scope_as = timur_ruler
			}
		}

		scope:timur_ruler = {
			add_character_flag = phx_timur_character
			if = {
				limit = { NOT = { has_trait = conqueror } }
				add_trait = conqueror
			}
			if = {
				limit = { NOT = { has_trait = strategist } }
				add_trait = strategist
			}
			if = {
				limit = { NOT = { has_character_modifier = phx_timur_scourge_modifier } }
				add_character_modifier = {
					modifier = phx_timur_scourge_modifier
					years = 35
				}
			}
			add_gold = 1200
			add_prestige = 1500
		}

		# If c_samarkand is held by AI (and not already Timur), grant Timur c_samarkand + k_eu5_tim
		if = {
			limit = {
				root = {
					is_ai = yes
					NOT = { this = scope:timur_ruler }
				}
			}
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = timur_rise_change
				add_claim_on_loss = no
			}
			title:c_samarkand = {
				change_title_holder = {
					holder = scope:timur_ruler
					change = scope:timur_rise_change
				}
			}
			if = {
				limit = { NOT = { exists = title:k_eu5_tim.holder } }
				title:k_eu5_tim = {
					change_title_holder = {
						holder = scope:timur_ruler
						change = scope:timur_rise_change
					}
				}
			}
			resolve_title_and_vassal_change = scope:timur_rise_change
		}

		# Spawn Timur's Gurkani host and launch story_phx_timur_rise if landed
		if = {
			limit = { scope:timur_ruler = { is_landed = yes } }
			scope:timur_ruler = {
				spawn_army = {
					name = phx_timur_gurkani_vanguard_name
					location = scope:samarkand_province
					men_at_arms = {
						type = heavy_horse_archers
						men = 800
					}
					men_at_arms = {
						type = horse_archers
						men = 600
					}
					men_at_arms = {
						type = armored_horsemen
						men = 400
					}
					levies = 3000
					inheritable = no
				}
				create_story = {
					type = story_phx_timur_rise
				}
			}
		}

		# O(1) notifications to Chagatai, Ilkhanate, Delhi, Golden Horde, Ottomans, and Samarkand's player holder
		if = {
			limit = {
				root = {
					is_ai = no
					NOT = { this = scope:timur_ruler }
				}
			}
			root = { trigger_event = { id = phx_glb.0021 days = 5 } }
		}
		if = {
			limit = {
				exists = title:e_chagatai.holder
				title:e_chagatai.holder = {
					is_alive = yes
					NOT = { this = scope:timur_ruler }
					NOT = { this = root }
				}
			}
			title:e_chagatai.holder = { trigger_event = { id = phx_glb.0021 days = 10 } }
		}
		if = {
			limit = {
				exists = title:e_ilkhanate.holder
				title:e_ilkhanate.holder = {
					is_alive = yes
					NOT = { this = scope:timur_ruler }
				}
			}
			title:e_ilkhanate.holder = { trigger_event = { id = phx_glb.0021 days = 15 } }
		}
		if = {
			limit = {
				exists = title:e_hindustan.holder
				title:e_hindustan.holder = {
					is_alive = yes
					NOT = { this = scope:timur_ruler }
				}
			}
			title:e_hindustan.holder = { trigger_event = { id = phx_glb.0021 days = 20 } }
		}
		if = {
			limit = {
				exists = title:e_golden_horde.holder
				title:e_golden_horde.holder = {
					is_alive = yes
					NOT = { this = scope:timur_ruler }
				}
			}
			title:e_golden_horde.holder = { trigger_event = { id = phx_glb.0021 days = 20 } }
		}
	}
}
