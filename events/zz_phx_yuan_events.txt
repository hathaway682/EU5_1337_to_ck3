					is_married = no
					is_betrothed = no
				}

				limit = {
					is_adult = yes

				}
				marry = scope:lady_ma
			}
			else = {
				create_betrothal = scope:lady_ma
			}
		}
		if = {
			limit = { exists = scope:guo_zixing }
			scope:guo_zixing = {

					target = root

	}
}

# 21. 0305 — Carving a Base in Huaisi (Fengyang / Chuzhou Uprising)
phx_yuan.0305 = {
	type = character_event
	title = phx_yuan.0305.t
	desc = phx_yuan.0305.desc
	theme = war
	override_background = { reference = tgp_camp_asia }
	left_portrait = {
		character = root
		animation = marshal
	}

	immediate = {
		set_global_variable = phx_red_turban_unrest_begun
		if = {
			limit = { exists = capital_province }
			capital_province = { save_scope_as = phx_zhu_loc }
		}
		else_if = {
			limit = { exists = domicile.domicile_location }
			domicile.domicile_location = { save_scope_as = phx_zhu_loc }
		}
		else = {
			title:c_fengyang.title_province = { save_scope_as = phx_zhu_loc }
		}
		if = {
			limit = {
				title:c_fengyang.holder ?= {
					NOT = { this = root }
					top_liege = {
						NOT = { this = root }
					}
				}
			}
			title:c_fengyang.holder.top_liege = { save_scope_as = phx_fengyang_enemy }
		}
	}

	# Option A: Strike Fengyang & Chuzhou in an open liberation uprising
	option = {
		name = phx_yuan.0305.a
		if = {
			limit = { NOT = { has_character_modifier = phx_rtr_red_turban_zeal_modifier } }
			add_character_modifier = {
				modifier = phx_rtr_red_turban_zeal_modifier
				years = 20
			}
		}
		spawn_army = {
			name = phx_rtr_huaisi_vanguard_name
			location = scope:phx_zhu_loc
			men_at_arms = {
				type = light_footmen
				men = 500
			}
			men_at_arms = {
				type = crossbowmen
				men = 300
			}
			men_at_arms = {
				type = fire_lancers
				men = 200
			}
			levies = 2000
			inheritable = no
		}
		if = {
			limit = {
				exists = scope:phx_fengyang_enemy
				NOT = { is_at_war_with = scope:phx_fengyang_enemy }
				can_declare_war = {
					defender = scope:phx_fengyang_enemy
					casus_belli = phx_rtr_uprising_seizure_cb
					target_titles = { title:c_fengyang }
				}
			}
			start_war = {
				cb = phx_rtr_uprising_seizure_cb
				target = scope:phx_fengyang_enemy
				target_title = title:c_fengyang
			}
		}
		else = {
			add_pressed_claim = title:c_fengyang
			add_pressed_claim = title:c_chuzhou
		}
		ai_chance = { base = 85 }
	}

	# Option B: Recruit quietly and prepare for crossing the Yangtze
	option = {
		name = phx_yuan.0305.b
		add_gold = 200
		add_prestige = 150
		add_pressed_claim = title:c_fengyang
		add_pressed_claim = title:c_yingtian
		ai_chance = { base = 15 }
	}
}
