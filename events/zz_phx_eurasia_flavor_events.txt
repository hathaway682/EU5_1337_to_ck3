# 0010 — 1350+ Crisis: Blood of Burgos (Castilian Succession — Pedro vs Enrique)
phx_eur.0010 = {
	type = character_event
	title = phx_eur.0010.t
	desc = phx_eur.0010.desc
	theme = intrigue
	override_background = { reference = corridor_night }
	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		trigger = { exists = scope:enrique_rival }
		character = scope:enrique_rival
		animation = anger
	}

	immediate = {
		if = {
			limit = {
				character:cas_enrique_ii_trastamara ?= {
					is_alive = yes
					NOT = { this = root }
				}
			}
			character:cas_enrique_ii_trastamara = { save_scope_as = enrique_rival }
		}
	}

	# Option A: Uphold strict royal prerogative and break rebellious magnates
	option = {
		name = phx_eur.0010.a
		add_dread = 25
		if = { limit = { is_valid_for_legitimacy_change = yes } add_legitimacy = 100 }
		if = {
			limit = { exists = scope:enrique_rival }
			scope:enrique_rival = {
				add_opinion = {
					target = root
					modifier = insulted_opinion
					opinion = -40
				}
			}
		}
		ai_chance = { base = 55 }
	}

	# Option B: Reconcile the noble houses with Cortes charters
	option = {
		name = phx_eur.0010.b
		remove_short_term_gold = 100
		add_prestige = -50
		if = {
			limit = { exists = scope:enrique_rival }
			scope:enrique_rival = {
				add_opinion = {
					target = root
					modifier = grateful_opinion
					opinion = 35
				}
			}
		}
		ai_chance = { base = 45 }
	}
}
