eu5_colonial_exploration.0002 = {
 hidden = yes
 scope = landed_title
 immediate = {
  if = {
   limit = { has_variable = eu5_voyage_delay var:eu5_voyage_delay > 0 }
   trigger_event = { id = eu5_colonial_exploration.0002 days = 5 }
   change_variable = { name = eu5_voyage_delay add = -5 }
  }
  else = {
   if = {
    limit = { has_variable = eu5_voyage_captain NOT = { var:eu5_voyage_captain ?= { is_alive = yes } } }
    set_variable = eu5_exploration_failed
   }
   if = {
    limit = { var:eu5_voyage_captain ?= { OR = { var:aod_v_crew <= 0 var:aod_v_morale <= 0 } } }
    set_variable = eu5_exploration_failed
   }
   var:eu5_voyage_captain ?= { remove_variable = eu5_exploration_command }
   remove_variable = eu5_voyage_captain
   remove_variable = eu5_voyage_delay
   if = {
    limit = { has_variable = eu5_exploration_failed }
    remove_variable = eu5_exploration_failed
    if = { limit = { has_variable = eu5_exploring_europe } remove_variable = eu5_exploring_europe }
    if = { limit = { has_variable = eu5_exploring_africa } remove_variable = eu5_exploring_africa }
    if = { limit = { has_variable = eu5_exploring_asia } remove_variable = eu5_exploring_asia }
    if = { limit = { has_variable = eu5_exploring_north_america } remove_variable = eu5_exploring_north_america }
    if = { limit = { has_variable = eu5_exploring_south_america } remove_variable = eu5_exploring_south_america }
    if = { limit = { has_variable = eu5_exploring_oceania } remove_variable = eu5_exploring_oceania }
    if = { limit = { has_variable = eu5_exploring_atlantic } remove_variable = eu5_exploring_atlantic }
    remove_variable = eu5_exploration_underway
    holder ?= { trigger_event = eu5_colonial_exploration.0004 }
   }
   else = {
    if = { limit = { has_variable = eu5_exploring_europe } remove_variable = eu5_exploring_europe set_variable = eu5_charted_europe }
    if = { limit = { has_variable = eu5_exploring_africa } remove_variable = eu5_exploring_africa set_variable = eu5_charted_africa }
    if = { limit = { has_variable = eu5_exploring_asia } remove_variable = eu5_exploring_asia set_variable = eu5_charted_asia }
    if = { limit = { has_variable = eu5_exploring_north_america } remove_variable = eu5_exploring_north_america set_variable = eu5_charted_north_america }
    if = { limit = { has_variable = eu5_exploring_south_america } remove_variable = eu5_exploring_south_america set_variable = eu5_charted_south_america }
    if = { limit = { has_variable = eu5_exploring_oceania } remove_variable = eu5_exploring_oceania set_variable = eu5_charted_oceania }
    if = { limit = { has_variable = eu5_exploring_atlantic } remove_variable = eu5_exploring_atlantic set_variable = eu5_charted_atlantic }
    remove_variable = eu5_exploration_underway
    save_scope_as = eu5_chart_source_title
    holder ?= {
     eu5_sync_realm_charts_effect = yes
     trigger_event = eu5_colonial_exploration.0003
    }
   }
  }
 }
}

# Adapted AoD voyage events: at sea.
eu5_colonial_exploration.0010 = {
 hidden = yes
 scope = landed_title
 immediate = {
  if = {
   limit = { has_variable = eu5_exploration_underway NOT = { has_variable = eu5_exploration_failed } exists = holder var:eu5_voyage_captain ?= { is_alive = yes } }
   save_scope_as = eu5_voyage_title
   var:eu5_voyage_captain = { save_scope_as = nwt_captain aod_vs_sea_tick_effect = yes }
   if = {
    limit = { scope:nwt_captain = { var:aod_v_crew <= 0 } }
    holder = { nwt_voyage_lost_effect = { REASON = death_disappearance } }
   }
   else_if = {
    limit = { scope:nwt_captain = { var:aod_v_morale <= 15 } }
    holder = { trigger_event = nwt_voyage.32 }
   }
   else = {
    holder = { random = { chance = 65 random_list = {
     16 = { trigger_event = nwt_voyage.30 }
     12 = { trigger_event = nwt_voyage.31 }
     10 = { trigger_event = nwt_voyage.33 }
     12 = { trigger_event = nwt_voyage.80 }
     10 = { trigger_event = nwt_voyage.81 }
     8 = { trigger_event = nwt_voyage.82 }
     10 = { trigger_event = nwt_voyage.83 }
     12 = { trigger_event = nwt_voyage.84 }
     10 = { trigger_event = nwt_voyage.85 }
     8 = { trigger_event = nwt_voyage.86 }
    } } }
   }
  }
 }
}
