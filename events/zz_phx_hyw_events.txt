	}

		if = {
			limit = {
				exists = title:k_england.holder

			}
			title:k_england.holder = { save_scope_as = english_king }
		}
	}

	# Option A: Declare Aquitaine forfeit and march on Guyenne at once!
	option = {
		name = phx_hyw.0010.a
		add_prestige = 150
		add_legitimacy = 100
		if = {
			limit = {
				exists = scope:english_king
				top_liege = this
				scope:english_king = { top_liege = this }
				NOT = { is_at_war_with = scope:english_king }
				NOT = { is_allied_to = scope:english_king }
				title:d_guyenne = {
					any_in_de_jure_hierarchy = {
						tier = tier_county
						holder ?= {
							OR = {
								this = scope:english_king
								is_vassal_or_below_of = scope:english_king
							}
						}
					}
				}
			}
			start_war = {
				cb = phx_hyw_aquitaine_confiscation_cb
				target = scope:english_king
				target_title = title:d_guyenne
			}
		}
		else = {
			custom_tooltip = phx_hyw_aquitaine_forfeit_cb_tt
			hidden_effect = {
				add_character_flag = phx_hyw_free_aquitaine_cb
			}
		}
		hidden_effect = {
			set_global_variable = phx_hyw_aquitaine_confiscated
			set_global_variable = phx_hyw_war_has_begun
			if = {
				limit = {
					exists = scope:english_king
					NOT = { has_global_variable = phx_hyw_edward_claim_event_fired }
				}
				set_global_variable = phx_hyw_edward_claim_event_fired
				scope:english_king = {
					trigger_event = {
						id = phx_hyw.0011
						days = 14
					}
				}
			}
		}
		ai_chance = { base = 60 }
	}

	# Option B: Issue the royal decree of confiscation and summon the feudal host first
	option = {
		name = phx_hyw.0010.b
		add_prestige = 100
		add_character_modifier = {
			modifier = phx_hyw_valois_chivalry_modifier
			years = 15
		}
		custom_tooltip = phx_hyw_aquitaine_forfeit_cb_tt
		hidden_effect = {
			add_character_flag = phx_hyw_free_aquitaine_cb
			set_global_variable = phx_hyw_aquitaine_confiscated
			if = {
				limit = {
					exists = scope:english_king
					NOT = { has_global_variable = phx_hyw_edward_claim_event_fired }
				}
				set_global_variable = phx_hyw_edward_claim_event_fired
				scope:english_king = {
